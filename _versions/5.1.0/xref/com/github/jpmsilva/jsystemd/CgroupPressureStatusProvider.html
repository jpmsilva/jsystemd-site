17  package com.github.jpmsilva.jsystemd;
18  
19  import java.io.IOException;
20  import java.nio.file.Path;
21  import java.util.ArrayList;
22  import java.util.List;
23  import java.util.Locale;
24  import org.jspecify.annotations.Nullable;
25  
26  /**
27   * Implementation of {@link StatusProvider} that provides the pressure the control group of the process is under, such as
28   * {@code Cgroup pressure: 2.1% cpu; 0.0% memory; 12.3% io; 40.0% throttled}: how much the service is held back by a lack of CPU, memory or I/O, which
29   * neither its CPU usage nor its memory usage tell.
30   *
31   * <p>The pressure of each resource is the share of the last 10 seconds in which at least one task of the service was stalled waiting for it, as the
32   * kernel measures it (pressure stall information): runnable but waiting for a CPU, reclaiming memory or waiting for swapped pages, or waiting for I/O. A
33   * service with as many busy threads as CPUs is under no CPU pressure, while one with more is. Memory pressure that keeps growing means the service is about
34   * to reach its {@code MemoryHigh} or {@code MemoryMax}, even though its heap may still have room.
35   *
36   * <p>The throttling is the share of the enforcement periods of the CPU quota of the service, since the status was last read, in which the service used up
37   * its quota, and was kept from running until the next one. It is only reported when the service has a quota, set by systemd with {@code CPUQuota} in the
38   * service unit. A throttled service is slow while the system has idle CPUs, which shows as latency rather than as CPU usage.
39   *
40   * <p>Each part is left out when the kernel does not report it: the pressure requires a kernel with pressure stall information enabled, and is not
41   * accounted for the root control group. Only the unified hierarchy of control groups (cgroup v2) is supported, the one systemd uses on current Linux
42   * distributions, as the legacy one is deprecated. Nothing is reported otherwise, or when not running on Linux.
43   *
44   * @author Joao Silva
45   */
46  public class CgroupPressureStatusProvider implements StatusProvider {
47  
48    private static final List<String> RESOURCES = List.of("cpu", "memory", "io");
49  
50    // the control group of the process, or null if it is not in the unified hierarchy
51    @Nullable
52    private final ControlGroup cgroup;
53  
54    // the enforcement periods of the CPU quota, and those in which the control group was throttled, when the status was last read, or -1 if they were not
55    // reported; guarded by this
56    private long lastPeriods;
57    private long lastThrottled;
58  
59    /**
60     * Create a new CgroupPressureStatusProvider, for the control group the process is in when created, which reports the throttling since it was created
61     * on the first read.
62     */
63    public CgroupPressureStatusProvider() {
64      this(ControlGroup.PROC_SELF_CGROUP, ControlGroup.CGROUP_ROOT);
65    }
66  
67    CgroupPressureStatusProvider(Path procSelfCgroup, Path root) {
68      // every control group of the unified hierarchy has this file, whichever controllers are enabled for it
69      this.cgroup = ControlGroup.find(procSelfCgroup, root, "cgroup.controllers");
70      long[] throttling = cgroup == null ? null : throttling(cgroup);
71      this.lastPeriods = throttling == null ? -1 : throttling[0];
72      this.lastThrottled = throttling == null ? -1 : throttling[1];
73    }
74  
75    @Override
76    public synchronized String status() {
77      ControlGroup cgroup = this.cgroup;
78      if (cgroup == null) {
79        return "";
80      }
81      // the statuses of all providers are joined with commas, so the details are separated otherwise
82      List<String> details = new ArrayList<>();
83      for (String resource : RESOURCES) {
84        double pressure = pressure(cgroup, resource);
85        if (pressure >= 0) {
86          details.add(String.format(Locale.ROOT, "%.1f%% %s", pressure, resource));
87        }
88      }
89      long[] throttling = throttling(cgroup);
90      if (throttling != null) {
91        // the counters only grow, but the process may be moved to another control group, so the differences are kept positive
92        long periods = throttling[0] - lastPeriods;
93        long throttled = Math.max(0, throttling[1] - lastThrottled);
94        // without a quota, there are no enforcement periods
95        if (lastPeriods >= 0 && periods > 0) {
96          details.add(String.format(Locale.ROOT, "%.1f%% throttled", Math.min(100.0, 100.0 * throttled / periods)));
97        }
98        lastPeriods = throttling[0];
99        lastThrottled = throttling[1];
100     } else {
101       lastPeriods = -1;
102       lastThrottled = -1;
103     }
104     return details.isEmpty() ? "" : "Cgroup pressure: " + String.join("; ", details);
105   }
106 
107   // the share of the last 10 seconds in which some task was stalled on the resource, as a percentage, or -1 if it is not reported
108   private static double pressure(ControlGroup cgroup, String resource) {
109     try {
110       // such as "some avg10=0.18 avg60=0.03 avg300=0.00 total=299040577", followed by the same for "full"
111       for (String line : cgroup.read(resource + ".pressure").split("\n")) {
112         if (line.startsWith("some ")) {
113           for (String field : line.split(" ")) {
114             if (field.startsWith("avg10=")) {
115               return Double.parseDouble(field.substring("avg10=".length()));
116             }
117           }
118         }
119       }
120     } catch (IOException | RuntimeException e) {
121       // no pressure stall information: not enabled in the kernel, turned off for the control group, or the control group is gone
122       return -1;
123     }
124     return -1;
125   }
126 
127   // the enforcement periods of the CPU quota so far, and those in which the control group was throttled, or null if they are not reported
128   private static long @Nullable [] throttling(ControlGroup cgroup) {
129     long periods = -1;
130     long throttled = -1;
131     try {
132       // such as "nr_periods 1200", among other counters; only there when the CPU controller is enabled for the control group
133       for (String line : cgroup.read("cpu.stat").split("\n")) {
134         if (line.startsWith("nr_periods ")) {
135           periods = Long.parseLong(line.substring("nr_periods ".length()).trim());
136         } else if (line.startsWith("nr_throttled ")) {
137           throttled = Long.parseLong(line.substring("nr_throttled ".length()).trim());
138         }
139       }
140     } catch (IOException | RuntimeException e) {
141       return null;
142     }
143     return periods < 0 || throttled < 0 ? null : new long[] {periods, throttled};
144   }
145 }
