17  package com.github.jpmsilva.jsystemd.keycloak;
18  
19  import static java.util.Objects.requireNonNull;
20  
21  import com.github.jpmsilva.jsystemd.DelayableHealthProvider;
22  import com.github.jpmsilva.jsystemd.HealthProvider;
23  import com.github.jpmsilva.jsystemd.keycloak.HealthReadiness.Checks;
24  import java.time.Duration;
25  import java.util.List;
26  import java.util.Map;
27  import org.jspecify.annotations.Nullable;
28  
29  /**
30   * A {@link HealthProvider} for the systemd watchdog that reports Keycloak as unhealthy while any of its readiness health checks is down, those of its
31   * {@code /health/ready} endpoint, which a load balancer stops sending requests on.
32   *
33   * <p>The checks are run periodically on a thread of their own, once started, and the health is the outcome of their last run: they may wait for what
34   * they check, such as a connection to the database, for longer than the watchdog allows between two heartbeats, and the heartbeat is only sent once
35   * the health providers have answered, so running them from there could have systemd restart Keycloak however long the unhealthy state is delayed for.
36   * Keycloak is also unhealthy while a run of the checks does not complete in time.
37   *
38   * <p>The checks mostly depend on what a restart of Keycloak does not fix, such as its database, so the unhealthy state should be delayed, with
39   * {@link #setUnhealthyDelay(Duration)}, for longer than the outages Keycloak is expected to ride out.
40   *
41   * @author Joao Silva
42   */
43  final class ReadinessHealthProvider extends DelayableHealthProvider {
44  
45    private final Checks checks;
46    private final Duration period;
47    private final Duration stuckAfter;
48  
49    // the outcome of the last run of the checks: null if they are all up, or else the names of those that are down, none if they could not run
50    @Nullable
51    private volatile List<String> down;
52  
53    // when the run of the checks that is under way started, as a value of nanoTime, or null between runs
54    @Nullable
55    private volatile Long running;
56  
57    // guarded by this
58    @Nullable
59    private Thread thread;
60  
61    /**
62     * Creates a new instance, which reports Keycloak as healthy until started.
63     *
64     * @param checks the readiness health checks of Keycloak
65     * @param period how long to wait between two runs of the checks
66     * @param stuckAfter how long a run of the checks may take before Keycloak is unhealthy
67     */
68    ReadinessHealthProvider(Checks checks, Duration period, Duration stuckAfter) {
69      this.checks = requireNonNull(checks, "Checks must not be null");
70      this.period = requireNonNull(period, "Period must not be null");
71      this.stuckAfter = requireNonNull(stuckAfter, "Stuck after must not be null");
72    }
73  
74    /**
75     * Starts running the checks periodically, once Keycloak is ready. Does nothing if already started.
76     */
77    synchronized void start() {
78      if (thread == null) {
79        Thread thread = new Thread(this::run, "jsystemd-readiness");
80        thread.setDaemon(true);
81        thread.start();
82        this.thread = thread;
83      }
84    }
85  
86    /**
87     * Stops running the checks, once Keycloak is stopping. Does nothing if not started, and the checks cannot be started again.
88     */
89    synchronized void stop() {
90      Thread thread = this.thread;
91      if (thread != null) {
92        thread.interrupt();
93      }
94    }
95  
96    private void run() {
97      try {
98        while (!Thread.currentThread().isInterrupted()) {
99          running = System.nanoTime();
100         try {
101           down = checks.down();
102         } catch (RuntimeException e) {
103           // the checks could not run, which leaves Keycloak unhealthy until they do
104           down = List.of();
105         } finally {
106           running = null;
107         }
108         Thread.sleep(period.toMillis());
109       }
110     } catch (InterruptedException e) {
111       Thread.currentThread().interrupt();
112     }
113   }
114 
115   /**
116    * Provides the health of Keycloak, without the delay: unhealthy while any readiness health check was down the last time they ran, or while they have
117    * been running for too long.
118    *
119    * @return the health of Keycloak
120    */
121   @Override
122   protected Health evaluateHealth() {
123     Long running = this.running;
124     if (running != null) {
125       // values of nanoTime are only compared by their difference, as they may overflow
126       Duration runningFor = Duration.ofNanos(System.nanoTime() - running);
127       if (runningFor.compareTo(stuckAfter) > 0) {
128         return new Health(false, Map.of("readinessHealthChecks", "running for " + runningFor.toSeconds() + " seconds"));
129       }
130     }
131     List<String> down = this.down;
132     if (down == null) {
133       return Health.HEALTHY;
134     }
135     return new Health(false, down.isEmpty() ? Map.of("readinessHealthChecks", "could not run") : Map.of("readinessHealthChecksDown", down));
136   }
137 }
