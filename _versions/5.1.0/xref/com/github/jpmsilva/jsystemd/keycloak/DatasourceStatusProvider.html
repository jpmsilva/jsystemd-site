37  /**
38   * Provides information regarding the connection pool of the datasource of Keycloak, the one of its database: as a {@link StatusProvider}, the
39   * connections in use out of the most the pool opens, and the threads waiting for a connection, such as {@code Datasource: 7/100 connections, 3 waiting},
40   * and as a {@link HealthProvider}, for the systemd watchdog, reporting Keycloak as unhealthy while the pool is starved, if enabled.
41   *
42   * <p>The pool only counts its connections when Keycloak is built with its metrics enabled ({@code metrics-enabled}): nothing is reported otherwise,
43   * and Keycloak is then always healthy.
44   *
45   * <p>The health is enabled on its own, as it lets systemd restart Keycloak: while disabled, Keycloak is always healthy. When enabled, it is unhealthy
46   * while the pool is starved: all its connections are in use, and threads are waiting for one, reporting both in the health details. A starved pool
47   * usually means connections leak, or are all held by requests that hang, which only a restart recovers from. As a burst of requests can also starve the
48   * pool for a while, the unhealthy state should be delayed, with {@link #setUnhealthyDelay(java.time.Duration)}: the pool must then stay starved, every
49   * time the health is evaluated, for that long.
50   *
51   * <p>A database that cannot be reached does not starve the pool: its threads wait for a connection, but the connections are not in use, as the pool
52   * cannot open them. That is left for the readiness health checks of Keycloak.
53   *
54   * @author Joao Silva
55   */
56  final class DatasourceStatusProvider extends DelayableHealthProvider implements StatusProvider {
57  
58    private static final Logger logger = LoggerFactory.getLogger(DatasourceStatusProvider.class);
59  
60    // looks up the datasource, which is null until Keycloak has created it
61    private final Supplier<@Nullable AgroalDataSource> lookup;
62  
63    private final boolean health;
64  
65    // set once found, read by the status updates and the health evaluations
66    @Nullable
67    private volatile AgroalDataSource dataSource;
68  
69    /**
70     * Creates a new instance.
71     *
72     * @param lookup looks up the datasource of Keycloak, returning null while there is none
73     * @param health <code>true</code> to report Keycloak as unhealthy while the pool is starved
74     */
75    DatasourceStatusProvider(Supplier<@Nullable AgroalDataSource> lookup, boolean health) {
76      this.lookup = requireNonNull(lookup, "Lookup must not be null");
77      this.health = health;
78    }
79  
80    // for tests: whether a starved pool reports Keycloak as unhealthy
81    boolean healthEnabled() {
82      return health;
83    }
84  
85    /**
86     * Looks up the datasource of Keycloak, its default one.
87     *
88     * @return the datasource, or null when there is none, as outside of Quarkus, as in tests
89     */
90    static @Nullable AgroalDataSource lookup() {
91      try {
92        ArcContainer container = Arc.container();
93        // a lookup of a bean that is not there would warn, unlike asking if it is there
94        if (container == null || container.beanManager().getBeans(AgroalDataSource.class).isEmpty()) {
95          return null;
96        }
97        InstanceHandle<AgroalDataSource> handle = container.instance(AgroalDataSource.class);
98        return handle.isAvailable() ? handle.get() : null;
99      } catch (LinkageError e) {
100       // a Keycloak without Agroal or ArC, which all versions so far have
101       logger.debug("Not reporting the datasource, as it cannot be looked up", e);
102       return null;
103     }
104   }
105 
106   // the datasource, when it counts its connections
107   private synchronized @Nullable AgroalDataSource counting() {
108     AgroalDataSource dataSource = this.dataSource;
109     if (dataSource == null) {
110       dataSource = lookup.get();
111       if (dataSource == null) {
112         return null;
113       }
114       this.dataSource = dataSource;
115       if (health && !dataSource.getConfiguration().metricsEnabled()) {
116         logger.warn("The health of the datasource is enabled, but its connection pool does not count its connections, so Keycloak is always healthy: "
117             + "build Keycloak with its metrics enabled (metrics-enabled=true)");
118       }
119     }
120     return dataSource.getConfiguration().metricsEnabled() ? dataSource : null;
121   }
122 
123   @Override
124   public String status() {
125     AgroalDataSource dataSource = counting();
126     if (dataSource == null) {
127       return "";
128     }
129     AgroalDataSourceMetrics metrics = dataSource.getMetrics();
130     return String.format(Locale.ROOT, "Datasource: %d/%d connections, %d waiting", metrics.activeCount(),
131         dataSource.getConfiguration().connectionPoolConfiguration().maxSize(), metrics.awaitingCount());
132   }
133 
134   /**
135    * Provides the health of Keycloak, without the delay: unhealthy while the pool is starved, if enabled.
136    *
137    * @return the health of Keycloak
138    */
139   @Override
140   protected Health evaluateHealth() {
141     AgroalDataSource dataSource = health ? counting() : null;
142     if (dataSource == null) {
143       return Health.HEALTHY;
144     }
145     AgroalDataSourceMetrics metrics = dataSource.getMetrics();
146     long active = metrics.activeCount();
147     int max = dataSource.getConfiguration().connectionPoolConfiguration().maxSize();
148     long awaiting = metrics.awaitingCount();
149     if (awaiting <= 0 || active < max) {
150       return Health.HEALTHY;
151     }
152     Map<String, Object> details = new LinkedHashMap<>();
153     details.put("activeConnections", active);
154     details.put("maxConnections", max);
155     details.put("threadsAwaitingConnection", awaiting);
156     return new Health(false, details);
157   }
158 }
