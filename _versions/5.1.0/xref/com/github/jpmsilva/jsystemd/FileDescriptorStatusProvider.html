17  package com.github.jpmsilva.jsystemd;
18  
19  import com.sun.management.UnixOperatingSystemMXBean;
20  import java.lang.management.ManagementFactory;
21  import java.util.LinkedHashMap;
22  import java.util.Locale;
23  import java.util.Map;
24  import org.jspecify.annotations.Nullable;
25  
26  /**
27   * Provides information regarding the file descriptors of the process: as a {@link StatusProvider}, the open file descriptors out of the most the
28   * process can open, such as {@code File descriptors: 812/65536}, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as
29   * unhealthy when they are about to run out, if enabled.
30   *
31   * <p>Every open file, socket and pipe takes a file descriptor, and once they run out, the application can no longer accept connections or open files.
32   * As a leak of file descriptors never resolves by itself, restarting the application is usually the only way out. The limit is the one systemd sets with
33   * {@code LimitNOFILE} in the service unit.
34   *
35   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
36   * unhealthy while the open file descriptors are at or above the given share of the limit, reporting both in the health details, under
37   * {@code openFileDescriptors} and {@code maxFileDescriptors}. As a burst of connections can also open many file descriptors for a short while, the
38   * unhealthy state can be delayed, with {@link #setUnhealthyDelay(java.time.Duration)}.
39   *
40   * <p>The file descriptors are only known on Unix-like systems, and nothing is reported otherwise.
41   *
42   * @author Joao Silva
43   */
44  public class FileDescriptorStatusProvider extends DelayableHealthProvider implements StatusProvider {
45  
46    @Nullable
47    private final UnixOperatingSystemMXBean system;
48  
49    private final boolean health;
50  
51    private final double threshold;
52  
53    /**
54     * Create a new FileDescriptorStatusProvider.
55     *
56     * @param health <code>true</code> to report the application as unhealthy when the file descriptors are about to run out
57     * @param threshold the share of the limit of file descriptors from which the application is unhealthy, such as <code>0.9</code> for 90%
58     * @throws IllegalArgumentException if the threshold is not greater than 0 and at most 1
59     */
60    public FileDescriptorStatusProvider(boolean health, double threshold) {
61      this(ManagementFactory.getOperatingSystemMXBean() instanceof UnixOperatingSystemMXBean system ? system : null, health, threshold);
62    }
63  
64    FileDescriptorStatusProvider(@Nullable UnixOperatingSystemMXBean system, boolean health, double threshold) {
65      if (!(threshold > 0 && threshold <= 1)) {
66        throw new IllegalArgumentException("Threshold must be greater than 0 and at most 1");
67      }
68      this.system = system;
69      this.health = health;
70      this.threshold = threshold;
71    }
72  
73    // for tests: whether the file descriptors about to run out report the application as unhealthy
74    boolean healthEnabled() {
75      return health;
76    }
77  
78    // for tests: the share of the limit from which the application is unhealthy
79    double threshold() {
80      return threshold;
81    }
82  
83    @Override
84    public String status() {
85      UnixOperatingSystemMXBean system = this.system;
86      if (system == null) {
87        return "";
88      }
89      return String.format(Locale.ROOT, "File descriptors: %d/%d", system.getOpenFileDescriptorCount(), system.getMaxFileDescriptorCount());
90    }
91  
92    /**
93     * Provides the health of the application, without the delay: unhealthy while the open file descriptors are at or above the threshold, if enabled.
94     *
95     * @return the health of the application
96     */
97    @Override
98    protected Health evaluateHealth() {
99      UnixOperatingSystemMXBean system = this.system;
100     if (system == null || !health) {
101       return Health.HEALTHY;
102     }
103     long open = system.getOpenFileDescriptorCount();
104     long max = system.getMaxFileDescriptorCount();
105     if (max <= 0 || open < threshold * max) {
106       return Health.HEALTHY;
107     }
108     Map<String, Object> details = new LinkedHashMap<>();
109     details.put("openFileDescriptors", open);
110     details.put("maxFileDescriptors", max);
111     return new Health(false, details);
112   }
113 }
