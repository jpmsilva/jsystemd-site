17  package com.github.jpmsilva.jsystemd;
18  
19  import java.lang.management.ManagementFactory;
20  import java.util.Locale;
21  import javax.management.JMException;
22  import javax.management.MBeanServer;
23  import javax.management.MalformedObjectNameException;
24  import javax.management.ObjectName;
25  import org.jspecify.annotations.Nullable;
26  
27  /**
28   * Implementation of {@link StatusProvider} that provides the virtual threads the JVM is running, such as
29   * {@code Virtual threads: 8 mounted; 340 queued; 8 carrier threads}: the virtual threads that are running, each mounted on a platform thread of the
30   * scheduler, its carrier, those that are ready to run but wait for a carrier, and the carrier threads the scheduler has.
31   *
32   * <p>The {@link ThreadStatusProvider} only reports platform threads, and thread pools mean little once the work runs on virtual threads, of which there
33   * may be any number. What limits them is their scheduler: virtual threads that keep waiting for a carrier, while every carrier has one mounted, are held
34   * back, either by more work than the CPUs can take, or by virtual threads that block without giving up their carrier, as when pinned.
35   *
36   * <p>Virtual threads that are blocked, waiting or sleeping are neither mounted nor queued, and are not counted: the JVM does not count them.
37   *
38   * <p>The scheduler is only observable on Java 24 and later: nothing is reported on earlier versions.
39   *
40   * @author Joao Silva
41   */
42  public class VirtualThreadStatusProvider implements StatusProvider {
43  
44    private static final String SCHEDULER = "jdk.management:type=VirtualThreadScheduler";
45  
46    private final MBeanServer server;
47  
48    // the management bean of the scheduler of the virtual threads, or null when the JVM has none
49    @Nullable
50    private final ObjectName scheduler;
51  
52    /**
53     * Create a new VirtualThreadStatusProvider.
54     */
55    public VirtualThreadStatusProvider() {
56      this(ManagementFactory.getPlatformMBeanServer());
57    }
58  
59    VirtualThreadStatusProvider(MBeanServer server) {
60      this.server = server;
61      this.scheduler = scheduler(server);
62    }
63  
64    // read by name, rather than through its interface, which the minimum supported version of Java does not have
65    private static @Nullable ObjectName scheduler(MBeanServer server) {
66      try {
67        ObjectName name = new ObjectName(SCHEDULER);
68        return server.isRegistered(name) ? name : null;
69      } catch (MalformedObjectNameException e) {
70        throw new IllegalStateException(e);
71      }
72    }
73  
74    @Override
75    public String status() {
76      ObjectName scheduler = this.scheduler;
77      if (scheduler == null) {
78        return "";
79      }
80      try {
81        return String.format(Locale.ROOT, "Virtual threads: %d mounted; %d queued; %d carrier threads",
82            ((Number) server.getAttribute(scheduler, "MountedVirtualThreadCount")).longValue(),
83            ((Number) server.getAttribute(scheduler, "QueuedVirtualThreadCount")).longValue(),
84            ((Number) server.getAttribute(scheduler, "PoolSize")).longValue());
85      } catch (JMException | RuntimeException e) {
86        // a scheduler that does not report them, such as one that is not the one of the JDK
87        return "";
88      }
89    }
90  }
