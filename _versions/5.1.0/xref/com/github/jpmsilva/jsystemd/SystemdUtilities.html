53    private static long readWatchdogUsec() {
54      String watchdogUsec = System.getenv("WATCHDOG_USEC");
55      if (watchdogUsec != null && !watchdogUsec.isEmpty()) {
56        try {
57          return Long.parseLong(watchdogUsec);
58        } catch (NumberFormatException e) {
59          logger.warn("Value of environment property WATCHDOG_USEC cannot be read as a number - watchdog disabled: {}", watchdogUsec);
60        }
61      }
62      return 0;
63    }

65    /**
66     * Allows determining if the process is running under systemd.
67     *
68     * <p>A process is said to be running under systemd if <ol> <li>the operating system name starts with {@code linux};</li> <li>the environment variable
69     * {@code NOTIFY_SOCKET} is set, and for a file system socket, the socket file exists (see {@link #isUsableNotifySocket(String)});</li> <li>the
70     * notifications could be set up (see {@link NotifyChannel}): on Java 22 and later, with the Foreign Function and Memory API, unless native access is
71     * denied, and otherwise, as on Java 17 to 21 or for {@code vsock:} addresses, the native libsystemd library could be loaded.</li> </ol>
72     *
73     * @return {@code true} if the process is running under systemd
74     */
75    static boolean isUnderSystemd() {
76      return SystemdNotify.usable();
77    }

124   /**
125    * Allows determining if a systemd notify socket address can be used.
126    *
127    * <p>Only file system sockets (absolute paths) can be checked for existence. Other addresses supported by systemd, such as sockets in the abstract
128    * namespace (starting with {@code @}) or {@code vsock:} addresses, have no file, and are left for the channel that sends the notifications to handle.
129    *
130    * @param address the contents of the environment variable {@code NOTIFY_SOCKET}
131    * @return {@code true} if the address is set and, for file system sockets, the socket file exists
132    */
133   static boolean isUsableNotifySocket(@Nullable String address) {
134     if (address == null || address.isEmpty()) {
135       return false;
136     }
137     return !address.startsWith("/") || Path.of(address).toFile().exists();
138   }

140   private static final String[] unitPrefixes = new String[]{"", "Ki", "Mi", "Gi", "Ti", "Pi", "Ei"};
141 
142   static String formatByteCount(long bytes) {
143     // Long.MIN_VALUE has no positive counterpart, so the largest positive value is used instead
144     final long abs = bytes == Long.MIN_VALUE ? Long.MAX_VALUE : Math.abs(bytes);
145 
146     int divider = 0;
147     int index = 0;
148     while (abs >> divider >= 1024) {
149       divider += 10;
150       index += 1;
151     }
152 
153     final double division = abs / (double) (1L << divider) * Math.signum(bytes);
154     // the root locale keeps the separators the same everywhere, such as "1,023.0 B", instead of "1.023,0 B" under a German locale
155     return String.format(Locale.ROOT, "%,.1f %sB", division, unitPrefixes[index]);
156   }

158   /**
159    * Creates an executor for periodic tasks, with a single daemon thread, so that it does not keep the JVM running. The thread is only started once a task
160    * is scheduled.
161    */
162   static ScheduledExecutorService newExecutor(String threadName) {
163     return Executors.newSingleThreadScheduledExecutor(r -> {
164       final Thread thread = new Thread(r, threadName);
165       thread.setDaemon(true);
166       return thread;
167     });
168   }
169 
170   /**
171    * Schedules a periodic task that survives exceptions: the executor would otherwise silently suppress all subsequent executions of a task that throws.
172    * The durations are truncated to whole microseconds, the resolution systemd uses for its own intervals.
173    *
174    * @throws IllegalArgumentException if the period is shorter than a microsecond
175    */
176   static ScheduledFuture<?> schedule(ScheduledExecutorService executor, String name, Runnable task,
177       Duration initialDelay, Duration period) {
178     return executor.scheduleAtFixedRate(() -> {
179       try {
180         task.run();
181       } catch (Exception e) {
182         logger.warn("Periodic {} failed", name, e);
183       }
184     }, MICROSECONDS.convert(initialDelay), MICROSECONDS.convert(period), MICROSECONDS);
185   }

187   /**
188    * Rejects null status providers when they are registered, rather than when they are later used, far from the code that registered them.
189    */
190   static List<StatusProvider> requireProviders(List<StatusProvider> providers) {
191     requireNonNull(providers, "Providers must not be null");
192     providers.forEach(provider -> requireNonNull(provider, "Providers must not contain null elements"));
193     return providers;
194   }

205   /**
206    * Rejects durations shorter than a microsecond, the resolution of systemd, as for the periods of the periodic tasks, and the amounts to extend the
207    * timeout by.
208    *
209    * @param duration the duration
210    * @param name what the duration is, for the message of the exception, such as {@code Period}
211    * @throws NullPointerException if the duration is {@code null}
212    * @throws IllegalArgumentException if the duration is shorter than a microsecond
213    */
214   static void requireAtLeastOneMicrosecond(Duration duration, String name) {
215     requireNonNull(duration, () -> name + " must not be null");
216     if (MICROSECONDS.convert(duration) < 1) {
217       throw new IllegalArgumentException(name + " must be of at least a microsecond, but was " + duration);
218     }
219   }
220 
221   /**
222    * Rejects negative durations, as for the delays before reporting an unhealthy state. A {@code null} duration, which stands for none, is accepted.
223    *
224    * @param duration the duration, or {@code null} for none
225    * @param name what the duration is, for the message of the exception, such as {@code Unhealthy delay}
226    * @throws IllegalArgumentException if the duration is negative
227    */
228   static void requireNotNegative(@Nullable Duration duration, String name) {
229     if (duration != null && duration.isNegative()) {
230       throw new IllegalArgumentException(name + " must not be negative, but was " + duration);
231     }
232   }

234   /**
235    * Waits for an executor that is shut down to terminate, for up to 10 seconds, before interrupting its running tasks.
236    *
237    * @throws InterruptedException if interrupted while waiting
238    */
239   static void awaitTermination(ScheduledExecutorService executor) throws InterruptedException {
240     if (!executor.awaitTermination(10, SECONDS)) {
241       executor.shutdownNow();
242     }
243   }
