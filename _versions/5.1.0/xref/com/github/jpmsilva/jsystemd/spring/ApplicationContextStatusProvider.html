17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.StartingStatusProvider;
20  import com.github.jpmsilva.jsystemd.StatusProvider;
21  import com.github.jpmsilva.jsystemd.Systemd;
22  import java.util.Arrays;
23  import java.util.Locale;
24  import java.util.Set;
25  import java.util.concurrent.TimeUnit;
26  import java.util.concurrent.atomic.AtomicLong;
27  import java.util.function.LongSupplier;
28  import java.util.function.Predicate;
29  import java.util.stream.Collectors;
30  import org.jspecify.annotations.Nullable;
31  import org.springframework.beans.BeansException;
32  import org.springframework.beans.factory.config.BeanDefinition;
33  import org.springframework.beans.factory.config.BeanPostProcessor;
34  import org.springframework.beans.factory.config.ConfigurableListableBeanFactory;
35  
36  /**
37   * Implementation of {@link StatusProvider} that provides information regarding the bean creation progress of a bean factory. It only reports while the
38   * application starts up, as a {@link StartingStatusProvider}.
39   *
40   * @author Joao Silva
41   * @see BeanPostProcessor
42   */
43  public class ApplicationContextStatusProvider implements StartingStatusProvider, BeanPostProcessor {
44  
45    /**
46     * The minimum interval between two notifications sent to systemd, as a bean is created far more often than systemd needs to hear about it. It is kept well
47     * below the amount by which the startup timeout is extended, so that the timeout keeps being extended while beans are being created.
48     */
49    static final long NOTIFY_INTERVAL_NANOS = TimeUnit.SECONDS.toNanos(1);
50  
51    private final ConfigurableListableBeanFactory factory;
52    private final LongSupplier nanoTime;
53    // when systemd was last notified, initially one interval in the past, so that the first bean created is notified right away
54    private final AtomicLong lastNotified;
55    // the names of the singletons to create, loaded when the first bean is created, and read by the periodic status update task too
56    @Nullable
57    private volatile Set<String> definitions;
58  
59    /**
60     * Creates a new instance, which sends the status to systemd as the beans are created.
61     *
62     * <p>The created instance is not registered as a status provider of {@link Systemd#instance()}: that is up to the caller.
63     *
64     * @param factory the bean factory of the application context, whose bean creation progress is reported
65     */
66    public ApplicationContextStatusProvider(ConfigurableListableBeanFactory factory) {
67      this(factory, System::nanoTime);
68    }
69  
70    /**
71     * Creates a new instance with the given source of time, for tests.
72     */
73    ApplicationContextStatusProvider(ConfigurableListableBeanFactory factory, LongSupplier nanoTime) {
74      this.factory = factory;
75      this.nanoTime = nanoTime;
76      this.lastNotified = new AtomicLong(nanoTime.getAsLong() - NOTIFY_INTERVAL_NANOS);
77    }
78  
79    private void ensureDefinitionsLoaded() {
80      if (definitions == null) {
81        definitions = Arrays.stream(factory.getBeanDefinitionNames())
82            .filter(isCreatedOnStartup(factory))
83            .collect(Collectors.toUnmodifiableSet());
84      }
85    }
86  
87    // the beans created eagerly while the context starts: neither abstract definitions, which are never created, nor lazy ones, which are only created once
88    // used, if ever, as with spring.main.lazy-initialization, so that the count reaches the total; merged, as a child definition inherits these from its parent
89    private Predicate<? super String> isCreatedOnStartup(ConfigurableListableBeanFactory factory) {
90      return beanName -> {
91        BeanDefinition definition = factory.getMergedBeanDefinition(beanName);
92        return definition.isSingleton() && !definition.isAbstract() && !definition.isLazyInit();
93      };
94    }
95  
96    /**
97     * Returns the bean creation progress, counting the singletons created so far, out of those created on startup: lazy ones, which may only be created
98     * once used, are left out, and nothing is reported when all of them are lazy. They are counted only when the status is read, that is, when systemd is
99     * notified (at most once a second while beans are being created) and on each periodic status update, rather than every time a bean is created.
100    */
101   @Override
102   public String status() {
103     Set<String> definitions = this.definitions;
104     if (definitions == null || definitions.isEmpty()) {
105       return "";
106     }
107     long created = definitions.stream().filter(factory::containsSingleton).count();
108     return String.format(Locale.ROOT, "Creating bean %d of %d", created, definitions.size());
109   }
110 
111   /**
112    * Notifies systemd of the bean creation progress, at most once a second, while the application starts up: the beans created afterwards, such as
113    * prototype or request scoped beans, are not part of the startup, and must neither extend the shutdown timeout once the application is stopping, nor
114    * update the status on the threads that create them, such as the threads that serve requests.
115    */
116   @Override
117   public @Nullable Object postProcessBeforeInitialization(Object bean, String beanName) throws BeansException {
118     // ready once stopping too, as the application may stop before it is ready
119     if (factory != null && !Systemd.instance().isReady()) {
120       ensureDefinitionsLoaded();
121       if (shouldNotify()) {
122         Systemd systemd = Systemd.instance();
123         systemd.extendTimeout();
124         systemd.updateStatus();
125       }
126     }
127     return bean;
128   }
129 
130   // claims the notification when the interval has passed, so that only one of several threads creating beans at the same time notifies systemd
131   private boolean shouldNotify() {
132     long now = nanoTime.getAsLong();
133     long last = lastNotified.get();
134     return now - last >= NOTIFY_INTERVAL_NANOS && lastNotified.compareAndSet(last, now);
135   }
136 
137   @Override
138   public @Nullable Object postProcessAfterInitialization(Object bean, String beanName) throws BeansException {
139     return bean;
140   }
141 }
