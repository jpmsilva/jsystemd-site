131   /**
132    * Event listener for the {@link AvailabilityChangeEvent} event to report to systemd that the service is ready.
133    *
134    * <p>The application is considered ready when the event is a {@link ReadinessState} with the state {@link ReadinessState#ACCEPTING_TRAFFIC}, unless
135    * configured to be ready once it has started instead, with <code>systemd.ready-event=application-started</code>, which
136    * {@link SystemdSpringApplicationRunListener} reports.
137    *
138    * @param event the {@link AvailabilityChangeEvent} received
139    */
140   @EventListener
141   public void started(@SuppressWarnings("unused") AvailabilityChangeEvent<ReadinessState> event) {
142     if (properties.getReadyEvent() == SystemdProperties.ReadyEvent.ACCEPTING_TRAFFIC && event.getState() == ReadinessState.ACCEPTING_TRAFFIC) {
143       Systemd.instance().ready();
144     }
145   }

401   /**
402    * Configuration class for systemd integration when using Micrometer, showing the values of chosen meters in the status.
403    */
404   @Configuration(proxyBeanMethods = false)
405   @ConditionalOnSystemd
406   @ConditionalOnClass(MeterRegistry.class)
407   @ConditionalOnProperty(prefix = SystemdProperties.STATUS_PROVIDERS_PREFIX, name = "micrometer.enabled")
408   public static class SystemdMicrometerConfiguration {
409 
410     SystemdMicrometerConfiguration() {
411     }
412 
413     /**
414      * The status provider of the meters, which finds them in the meter registry of the application: the primary one when there are several, as Spring Boot
415      * sets up, which measures with all the others. The registry is looked up when the bean is created, rather than required for it to be, so that the
416      * registry is found whatever the order of the autoconfigurations.
417      *
418      * @param registryProvider the meter registry of the application, if any
419      * @param properties the properties of the integration
420      * @return the status provider of the meters
421      */
422     @Bean
423     @Order(10000)
424     @ConditionalOnMissingBean
425     MicrometerStatusProvider systemdMicrometerStatusProvider(ObjectProvider<MeterRegistry> registryProvider, SystemdProperties properties) {
426       MeterRegistry registry = registryProvider.getIfUnique();
427       if (registry == null) {
428         logger.warn("There is no meter registry, or there are several and none is the primary one, so the meters are looked up in the global registry of "
429             + "Micrometer; check that the metrics autoconfiguration of Spring Boot is not excluded");
430         registry = Metrics.globalRegistry;
431       }
432       List<MicrometerStatusProvider.ShownMeter> meters = properties.getStatusProviders().getMicrometer().getMeters().stream()
433           .map(meter -> new MicrometerStatusProvider.ShownMeter(requireNonNull(meter.getName(), "Name must not be null"), meter.getTags(), meter.getLabel()))
434           .toList();
435       return new MicrometerStatusProvider(registry, meters);
436     }
437   }

628   /**
629    * Configuration class for systemd integration when using Spring Boot Actuator, providing a {@link HealthProvider} based on its health contributors.
630    */
631   @Configuration(proxyBeanMethods = false)
632   @ConditionalOnSystemd
633   @ConditionalOnClass(HealthIndicator.class)
634   @ConditionalOnProperty(prefix = SystemdProperties.STATUS_PROVIDERS_PREFIX, name = "actuator-health.enabled", matchIfMissing = true)
635   public static class SystemdActuatorHealthConfiguration {
636 
637     SystemdActuatorHealthConfiguration() {
638     }
639 
640     @Bean
641     @Order(-8000)
642     @ConditionalOnMissingBean(ActuatorHealthStatusProvider.class)
643     ActuatorHealthStatusProvider systemdActuatorHealthStatusProvider(ObjectProvider<HealthContributorRegistry> registryProvider,
644         ObjectProvider<ReactiveHealthContributorRegistry> reactiveRegistryProvider, ObjectProvider<HealthEndpointGroups> groupsProvider,
645         SystemdProperties systemdProperties) {
646       requireNonNull(registryProvider, "Registry provider must not be null");
647       requireNonNull(reactiveRegistryProvider, "Reactive registry provider must not be null");
648       requireNonNull(systemdProperties, "Properties must not be null");
649       SystemdProperties.ActuatorHealth properties = systemdProperties.getStatusProviders().getActuatorHealth();
650 
651       // the same contributors, with the same names, as the health endpoint, which also adapts the reactive ones
652       List<HealthContributors> contributors = new ArrayList<>();
653       HealthContributorRegistry registry = registryProvider.getIfAvailable();
654       if (registry == null) {
655         logger.warn("There is no health contributor registry, so the Spring Boot Actuator health is always reported as healthy; check that the health "
656             + "contributor registry autoconfiguration of Spring Boot is not excluded");
657       } else {
658         contributors.add(registry);
659       }
660       reactiveRegistryProvider.ifAvailable(reactiveRegistry -> contributors.add(reactiveRegistry.asHealthContributors()));
661       Set<Status> unhealthyStatusCodes = properties.getUnhealthyStatusCodes().stream().map(Status::new).collect(Collectors.toSet());
662       if (properties.isHealth() && unhealthyStatusCodes.isEmpty() && Systemd.watchdogTimeout().isPresent() && !systemdProperties.isWatchdogSuppressWarning()) {
663         logger.warn("The systemd watchdog is enabled but no health status codes are considered as unhealthy, so the watchdog heartbeat will never be "
664             + "suppressed; set '{}.unhealthy-status-codes' with the status codes to consider as unhealthy, or suppress this warning with "
665             + "'{}.watchdog-suppress-warning=true'", SystemdProperties.ActuatorHealth.PREFIX, SystemdProperties.PREFIX);
666       }
667       ActuatorHealthStatusProvider provider = new ActuatorHealthStatusProvider(HealthContributors.of(contributors.toArray(HealthContributors[]::new)),
668           unhealthyStatusCodes, member(properties.getGroup(), groupsProvider), properties.isHealth());
669       provider.setUnhealthyDelay(properties.getUnhealthyDelay());
670       return provider;
671     }
672 
673     /**
674      * Returns the membership test of the health group with the given name, the one of the health endpoint, or one that takes all the health indicators into
675      * account without a group.
676      *
677      * @throws InvalidConfigurationPropertyValueException if there is no health group with the given name
678      */
679     private static Predicate<String> member(@Nullable String group, ObjectProvider<HealthEndpointGroups> groupsProvider) {
680       if (group == null) {
681         return name -> true;
682       }
683       String property = SystemdProperties.ActuatorHealth.PREFIX + ".group";
684       HealthEndpointGroups groups = groupsProvider.getIfAvailable();
685       if (groups == null) {
686         throw new InvalidConfigurationPropertyValueException(property, group, "There are no health groups, as the health endpoint of Spring Boot "
687             + "Actuator is not available: check that it is not disabled, nor its autoconfiguration excluded");
688       }
689       HealthEndpointGroup healthGroup = groups.get(group);
690       if (healthGroup == null) {
691         throw new InvalidConfigurationPropertyValueException(property, group, "There is no such health group; the health groups are " + groups.getNames()
692             + " (the liveness and readiness groups are not set up with management.endpoint.health.probes.enabled=false)");
693       }
694       return healthGroup::isMember;
695     }

698   /**
699    * Makes the systemd status include the status of the application, once all beans are created, so that the {@link StatusProvider} beans defined by
700    * the application are also found. The beans are added to the status providers of systemd, after the ones already registered (such as the startup status
701    * providers), in the order given by the <code>@Order</code> of their bean methods or classes. Beans already registered are not added again. While there
702    * is no status provider, no status is sent.
703    *
704    * <p>The built-in status providers, such as the {@link HeapStatusProvider}, are added while enabled, with the
705    * <code>systemd.status-providers.&lt;key&gt;.enabled</code> properties: most of them are enabled by default, and the others, such as the
706    * {@link GarbageCollectionStatusProvider} and the {@link CpuStatusProvider}, are opted into.
707    *
708    * @param statuses the status providers of the application
709    * @return the callback that adds the status providers to the systemd status
710    */
711   @Bean
712   SmartInitializingSingleton systemdStatusProviders(ObjectProvider<StatusProvider> statuses) {
713     return () -> {
714       Systemd systemd = Systemd.instance();
715       List<StatusProvider> registered = systemd.getStatusProviders();
716       systemd.addStatusProviders(statuses.orderedStream()
717           .filter(provider -> !registered.contains(provider))
718           .toList());
719     };
720   }

722   /**
723    * Makes the systemd watchdog take the application health into account, once all beans are created, so that the {@link HealthProvider} beans defined by
724    * the application are also found. The beans are added to the health providers of systemd, after the ones already registered, in the order given by the
725    * <code>@Order</code> of their bean methods or classes. Beans already registered are not added again. While there is no health provider, the watchdog
726    * heartbeat is always sent.
727    *
728    * <p>The built-in status providers that are also health providers, such as the {@link ThreadStatusProvider}, the {@link FileDescriptorStatusProvider},
729    * the {@link CgroupTasksStatusProvider}, the {@link GarbageCollectionStatusProvider}, the {@link HikariStatusProvider}, the
730    * {@link TomcatStatusProvider}, the {@link JettyStatusProvider}, the {@link AvailabilityStatusProvider} and the
731    * {@link ActuatorHealthStatusProvider}, are added while enabled. Their health conditions are controlled by their own properties, all disabled by
732    * default, so that they are always healthy unless opted into.
733    *
734    * @param healthProviders the health providers of the application
735    * @return the callback that adds the health providers to the watchdog
736    */
737   @Bean
738   SmartInitializingSingleton systemdHealthProviders(ObjectProvider<HealthProvider> healthProviders) {
739     return () -> {
740       Systemd systemd = Systemd.instance();
741       List<HealthProvider> registered = systemd.getHealthProviders();
742       systemd.addHealthProviders(healthProviders.orderedStream()
743           .filter(provider -> !registered.contains(provider))
744           .toList());
745     };
746   }
