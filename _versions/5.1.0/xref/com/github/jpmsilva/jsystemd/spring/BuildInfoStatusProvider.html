17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.StatusProvider;
20  import org.jspecify.annotations.Nullable;
21  import org.springframework.boot.info.BuildProperties;
22  import org.springframework.boot.info.GitProperties;
23  
24  /**
25   * Implementation of {@link StatusProvider} that provides the version of the application that is running, from the build information that Spring Boot
26   * exposes: the version of the build, followed by the short id of the commit it was built from, such as {@code Version: 1.4.2 (a1b2c3d)}.
27   *
28   * <p>The version of the build is the one of the {@link BuildProperties}, which Spring Boot reads from {@code META-INF/build-info.properties}, as generated
29   * by the {@code build-info} goal of its Maven plugin, or the {@code buildInfo()} task of its Gradle plugin. The commit is the one of the
30   * {@link GitProperties}, read from {@code git.properties}, as generated by the Git commit id plugins of Maven and Gradle. With only one of them, only the
31   * version, such as {@code Version: 1.4.2}, or only the commit, such as {@code Version: a1b2c3d}, is reported, and nothing is reported without both.
32   *
33   * <p>The version never changes while the application runs, so it is read once, when this provider is created.
34   *
35   * @author Joao Silva
36   */
37  public class BuildInfoStatusProvider implements StatusProvider {
38  
39    private final String status;
40  
41    /**
42     * Creates a new instance, which reports the version of the given build, and the commit it was built from.
43     *
44     * @param build the build information of the application, or <code>null</code> when it has none
45     * @param git the information of the commit the application was built from, or <code>null</code> when it has none
46     */
47    public BuildInfoStatusProvider(@Nullable BuildProperties build, @Nullable GitProperties git) {
48      this.status = status(present(build == null ? null : build.getVersion()), present(git == null ? null : git.getShortCommitId()));
49    }
50  
51    // a property that is there without a value is the same as one that is not
52    private static @Nullable String present(@Nullable String value) {
53      return value == null || value.isBlank() ? null : value.strip();
54    }
55  
56    private static String status(@Nullable String version, @Nullable String commit) {
57      if (version == null) {
58        return commit == null ? "" : "Version: " + commit;
59      }
60      return commit == null ? "Version: " + version : "Version: " + version + " (" + commit + ")";
61    }
62  
63    @Override
64    public String status() {
65      return status;
66    }
67  }
