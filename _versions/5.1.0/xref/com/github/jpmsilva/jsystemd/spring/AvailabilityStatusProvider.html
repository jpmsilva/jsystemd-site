17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.DelayableHealthProvider;
20  import com.github.jpmsilva.jsystemd.HealthProvider;
21  import com.github.jpmsilva.jsystemd.StatusProvider;
22  import java.util.ArrayList;
23  import java.util.List;
24  import java.util.Locale;
25  import java.util.Map;
26  import org.jspecify.annotations.Nullable;
27  import org.springframework.boot.availability.AvailabilityChangeEvent;
28  import org.springframework.boot.availability.AvailabilityState;
29  import org.springframework.boot.availability.LivenessState;
30  import org.springframework.boot.availability.ReadinessState;
31  import org.springframework.context.ApplicationListener;
32  
33  /**
34   * Provides the availability of the application, as Spring Boot reports it with its {@link AvailabilityChangeEvent}: as a {@link StatusProvider},
35   * whether it is live, and whether it is ready to accept traffic, such as {@code Liveness: correct; readiness: accepting traffic}, and as a
36   * {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy while its liveness is broken, if enabled.
37   *
38   * <p>Spring Boot reports the application as live and ready once it starts, and, for web applications, as refusing traffic once it starts shutting down.
39   * The application can also report a change of availability itself, such as {@link LivenessState#BROKEN} when it can no longer recover, with
40   * {@link AvailabilityChangeEvent#publish(org.springframework.context.ApplicationEventPublisher, Object, AvailabilityState)}.
41   *
42   * <p>Each state is only reported once it is first published.
43   *
44   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
45   * unhealthy while the liveness state is {@link LivenessState#BROKEN}, which Spring Boot defines as a state the application cannot recover from, so that
46   * restarting it is the way out: it is reported as unrecoverable, which triggers the watchdog right away, rather than once {@code WatchdogSec} elapses.
47   * The state is reported in the health details, under {@code livenessState}. Unlike the health indicators of Spring Boot Actuator, the liveness state does
48   * not depend on external systems. The unhealthy state can still be delayed, with {@link #setUnhealthyDelay(java.time.Duration)}, for applications that
49   * may report a broken liveness state and then recover from it: the watchdog is then only triggered once the delay elapses.
50   *
51   * @author Joao Silva
52   */
53  public class AvailabilityStatusProvider extends DelayableHealthProvider implements StatusProvider, ApplicationListener<AvailabilityChangeEvent<?>> {
54  
55    private final boolean health;
56  
57    // written by the event listener, read by the status updates
58    @Nullable
59    private volatile LivenessState liveness;
60  
61    @Nullable
62    private volatile ReadinessState readiness;
63  
64    /**
65     * Create a new AvailabilityStatusProvider.
66     *
67     * @param health <code>true</code> to report the application as unhealthy while its liveness state is broken
68     */
69    public AvailabilityStatusProvider(boolean health) {
70      this.health = health;
71    }
72  
73    // for tests: whether a broken liveness state reports the application as unhealthy
74    boolean healthEnabled() {
75      return health;
76    }
77  
78    /**
79     * Keeps the latest liveness and readiness states of the event. Other states are ignored.
80     *
81     * @param event the event of a change of availability
82     */
83    @Override
84    public void onApplicationEvent(AvailabilityChangeEvent<?> event) {
85      AvailabilityState state = event.getState();
86      if (state instanceof LivenessState livenessState) {
87        liveness = livenessState;
88      } else if (state instanceof ReadinessState readinessState) {
89        readiness = readinessState;
90      }
91    }
92  
93    @Override
94    public String status() {
95      List<String> details = new ArrayList<>();
96      LivenessState liveness = this.liveness;
97      if (liveness != null) {
98        details.add("Liveness: " + describe(liveness));
99      }
100     ReadinessState readiness = this.readiness;
101     if (readiness != null) {
102       details.add((details.isEmpty() ? "Readiness: " : "readiness: ") + describe(readiness));
103     }
104     return String.join("; ", details);
105   }
106 
107   /**
108    * Provides the health of the application, without the delay: unhealthy, and unrecoverable, while the liveness state is broken, if enabled.
109    *
110    * @return the health of the application
111    */
112   @Override
113   protected Health evaluateHealth() {
114     return health && liveness == LivenessState.BROKEN ? Health.unrecoverable(Map.of("livenessState", LivenessState.BROKEN.name())) : Health.HEALTHY;
115   }
116 
117   // the root locale keeps the text the same everywhere: under a Turkish locale, for instance, the "I" in "ACCEPTING" would become a dotless "ı"
118   private static String describe(Enum<?> state) {
119     return state.name().toLowerCase(Locale.ROOT).replace('_', ' ');
120   }
121 }
