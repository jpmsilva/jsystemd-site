41  /**
42   * Provides information regarding the embedded Jetty servers of the application: as a {@link StatusProvider}, their connections and threads, and as a
43   * {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy while all the threads of a server are taken and requests
44   * wait for one, if enabled.
45   *
46   * <p>For each server, named after the port it listens on, it reports the open connections, out of the maximum the server accepts when limited, and, when
47   * the requests are handled by a thread pool rather than by virtual threads, the threads running requests out of the threads in the pool, such as
48   * {@code jetty-8080: 12/1000 connections, 3/10 threads}. The threads Jetty keeps for itself, to accept connections and wait for their data, are left out of
49   * the threads running requests. A server that is not started is left out.
50   *
51   * <p>The servers are found as they start, from their {@link WebServerInitializedEvent}: the server of the application, as well as the server of the
52   * management endpoints when they use a port of their own. Nothing is reported until then.
53   *
54   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
55   * unhealthy while any server is saturated: its pool has all the threads it can have, the most it grows to rather than those it has so far, none of them
56   * is idle, and requests are waiting for one, naming the saturated servers in the health details, under {@code saturatedServers}. A server that stays
57   * saturated usually means that its requests hang, on something that never answers, which only a restart recovers from. As a burst of requests can also
58   * saturate a server for a while, the unhealthy state should be delayed, with {@link #setUnhealthyDelay(java.time.Duration)}: a server must then stay
59   * saturated, every time the health is evaluated, for that long.
60   *
61   * <p>The threads running requests, as shown in the status, do not tell if a server is saturated: Jetty may keep one of its threads in reserve, to wait
62   * for the data of the connections, rather than run a request with it, so that requests can be waiting while the status shows a thread to spare.
63   *
64   * <p>A server whose requests are handled by virtual threads has no thread pool to exhaust, and is never saturated.
65   *
66   * @author Joao Silva
67   */
68  public class JettyStatusProvider extends DelayableHealthProvider implements StatusProvider, ApplicationListener<WebServerInitializedEvent> {
69  
70    private final boolean health;
71  
72    // added by the event listener, read by the status updates and the health evaluations
73    private final CopyOnWriteArrayList<Server> servers = new CopyOnWriteArrayList<>();
74  
75    /**
76     * Create a new JettyStatusProvider, which always reports the application as healthy.
77     */
78    public JettyStatusProvider() {
79      this(false);
80    }
81  
82    /**
83     * Create a new JettyStatusProvider.
84     *
85     * @param health <code>true</code> to report the application as unhealthy while all the threads of a server are taken and requests wait for one
86     */
87    public JettyStatusProvider(boolean health) {
88      this.health = health;
89    }
90  
91    // for tests: whether a saturated server reports the application as unhealthy
92    boolean healthEnabled() {
93      return health;
94    }
95  
96    /**
97     * Adds the server of the event to the servers that are reported, if it is a Jetty server.
98     *
99     * @param event the event of a web server that started
100    */
101   @Override
102   public void onApplicationEvent(WebServerInitializedEvent event) {
103     if (event.getWebServer() instanceof JettyWebServer server) {
104       servers.addIfAbsent(server.getServer());
105     }
106   }
107 
108   @Override
109   public String status() {
110     // sorted by name, so that the order does not depend on the order in which the servers start
111     Map<String, String> statuses = new TreeMap<>();
112     for (Server server : servers) {
113       String name = name(server);
114       if (name != null) {
115         statuses.put(name, name + ": " + String.join(", ", details(server)));
116       }
117     }
118     return String.join(", ", statuses.values());
119   }
120 
121   /**
122    * Returns the name of the server, after the ports its connectors listen on, or <code>null</code> if none is listening.
123    */
124   private static @Nullable String name(Server server) {
125     List<String> ports = new ArrayList<>();
126     for (Connector connector : server.getConnectors()) {
127       // -1 while the connector is not started
128       if (connector instanceof NetworkConnector network && network.getLocalPort() > 0) {
129         ports.add(String.valueOf(network.getLocalPort()));
130       }
131     }
132     return ports.isEmpty() ? null : "jetty-" + String.join("+", ports);
133   }
134 
135   private static List<String> details(Server server) {
136     List<String> details = new ArrayList<>();
137     int connections = 0;
138     for (Connector connector : server.getConnectors()) {
139       if (connector instanceof AbstractConnector abstractConnector) {
140         connections += abstractConnector.getConnectedEndPoints().size();
141       }
142     }
143     // the maximum is only set with server.jetty.max-connections, which limits the connections of all the connectors of the server together
144     NetworkConnectionLimit limit = server.getBean(NetworkConnectionLimit.class);
145     details.add(limit != null ? String.format(Locale.ROOT, "%d/%d connections", connections, limit.getMaxNetworkConnectionCount())
146         : String.format(Locale.ROOT, "%d connections", connections));
147     // virtual threads are not pooled, so there is nothing to report on them; the threads leased by Jetty, the acceptors and selectors, never run requests
148     if (server.getThreadPool() instanceof QueuedThreadPool pool) {
149       details.add(String.format(Locale.ROOT, "%d/%d threads", pool.getUtilizedThreads(), pool.getThreads() - pool.getLeasedThreads()));
150     }
151     return details;
152   }
153 
154   /**
155    * Provides the health of the application, without the delay: unhealthy while all the threads of any server are taken and requests wait for one, if
156    * enabled.
157    *
158    * @return the health of the application
159    */
160   @Override
161   protected Health evaluateHealth() {
162     if (!health) {
163       return Health.HEALTHY;
164     }
165     // sorted by name, as the status
166     Set<String> saturated = new TreeSet<>();
167     for (Server server : servers) {
168       String name = name(server);
169       // the same thread pool as the status; virtual threads are not pooled, so a server that uses them is never saturated
170       if (name != null && server.getThreadPool() instanceof QueuedThreadPool pool && saturated(pool)) {
171         saturated.add(name);
172       }
173     }
174     return saturated.isEmpty() ? Health.HEALTHY : new Health(false, Map.of("saturatedServers", List.copyOf(saturated)));
175   }
176 
177   // requests are waiting for a thread, and none will come: a pool that can still grow starts one, and an idle one is about to take a request; the threads
178   // running requests are not compared to those the pool can have, as Jetty may keep one in reserve rather than run a request with it
179   private static boolean saturated(QueuedThreadPool pool) {
180     return pool.getQueueSize() > 0 && pool.getIdleThreads() == 0 && pool.getThreads() >= pool.getMaxThreads();
181   }
182 }
