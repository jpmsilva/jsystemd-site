42  /**
43   * Provides information regarding the R2DBC connection pools of the application, those of {@code r2dbc-pool}: as a {@link StatusProvider}, for each
44   * pool, named after its connection factory bean, the connections in use out of the most the pool opens, and the subscribers waiting for a connection,
45   * such as {@code connectionFactory: 7/10 connections, 3 waiting}, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as
46   * unhealthy while a pool is starved, if enabled.
47   *
48   * <p>The pools are those of the {@link ConnectionFactory} beans that are, or wrap, a {@link ConnectionPool}, as Spring Boot sets them up with
49   * {@code spring.r2dbc.pool.enabled}, found once all singleton beans are created, so that lazy ones are not created. A pool that is closed is left out.
50   *
51   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
52   * unhealthy while any pool is starved: all its connections are in use, and subscribers are waiting for one, naming the starved connection factories in
53   * the health details, under {@code starvedConnectionFactories}. A starved pool usually means connections leak, as code that takes a connection never
54   * gives it back, which only a restart recovers from. As a burst of requests can also starve a pool for a while, the unhealthy state should be delayed,
55   * with {@link #setUnhealthyDelay(java.time.Duration)}: the pool must then stay starved, every time the health is evaluated, for that long.
56   *
57   * <p>A database that cannot be reached does not starve the pools: their subscribers wait for a connection, but the connections are not in use, as the
58   * pools cannot open them. That is left for the health indicators of Spring Boot Actuator.
59   *
60   * @author Joao Silva
61   */
62  public class R2dbcPoolStatusProvider extends DelayableHealthProvider implements StatusProvider, SmartInitializingSingleton {
63  
64    private final ConfigurableListableBeanFactory beanFactory;
65  
66    private final boolean health;
67  
68    // set once all singleton beans are created, read by the status updates and the health evaluations
69    private volatile Map<String, ConnectionPool> pools = Map.of();
70  
71    /**
72     * Creates a new instance that finds the connection factories in the given bean factory.
73     *
74     * @param beanFactory the bean factory to find the connection factories in
75     * @param health <code>true</code> to report the application as unhealthy while any pool is starved
76     */
77    public R2dbcPoolStatusProvider(ConfigurableListableBeanFactory beanFactory, boolean health) {
78      this.beanFactory = requireNonNull(beanFactory, "Bean factory must not be null");
79      this.health = health;
80    }
81  
82    // for tests: whether a starved pool reports the application as unhealthy
83    boolean healthEnabled() {
84      return health;
85    }
86  
87    /**
88     * Finds the R2DBC connection pools among the connection factory beans that are already created.
89     */
90    @Override
91    public void afterSingletonsInstantiated() {
92      Map<String, ConnectionPool> found = new LinkedHashMap<>();
93      // a pool wrapped by several connection factory beans, such as a proxy that is also a bean, is only reported once, under the first name
94      Set<ConnectionPool> seen = Collections.newSetFromMap(new IdentityHashMap<>());
95      createdBeans(beanFactory, ConnectionFactory.class).forEach((name, connectionFactory) -> {
96        ConnectionPool pool = unwrap(connectionFactory);
97        if (pool != null && seen.add(pool)) {
98          found.put(name, pool);
99        }
100     });
101     pools = Collections.unmodifiableMap(found);
102   }
103 
104   /**
105    * Returns the connection pool that the given connection factory is, or wraps, as Spring Boot does for the metrics of the pools.
106    */
107   private static @Nullable ConnectionPool unwrap(@Nullable Object candidate) {
108     if (candidate instanceof ConnectionPool pool) {
109       return pool;
110     }
111     return candidate instanceof Wrapped<?> wrapped ? unwrap(wrapped.unwrap()) : null;
112   }
113 
114   /**
115    * Reads the metrics of each pool that is open.
116    */
117   private void forEachPool(BiConsumer<String, PoolMetrics> action) {
118     pools.forEach((name, pool) -> {
119       if (!pool.isDisposed()) {
120         pool.getMetrics().ifPresent(metrics -> action.accept(name, metrics));
121       }
122     });
123   }
124 
125   @Override
126   public String status() {
127     List<String> statuses = new ArrayList<>();
128     forEachPool((name, metrics) -> statuses.add(String.format(Locale.ROOT, "%s: %d/%d connections, %d waiting", name, metrics.acquiredSize(),
129         metrics.getMaxAllocatedSize(), metrics.pendingAcquireSize())));
130     return String.join(", ", statuses);
131   }
132 
133   /**
134    * Provides the health of the application, without the delay: unhealthy while any pool is starved, if enabled.
135    *
136    * @return the health of the application
137    */
138   @Override
139   protected Health evaluateHealth() {
140     if (!health) {
141       return Health.HEALTHY;
142     }
143     List<String> starved = new ArrayList<>();
144     forEachPool((name, metrics) -> {
145       if (metrics.pendingAcquireSize() > 0 && metrics.acquiredSize() >= metrics.getMaxAllocatedSize()) {
146         starved.add(name);
147       }
148     });
149     return starved.isEmpty() ? Health.HEALTHY : new Health(false, Map.of("starvedConnectionFactories", starved));
150   }
151 }
