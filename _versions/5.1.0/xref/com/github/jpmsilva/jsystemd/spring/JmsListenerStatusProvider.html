17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.HealthProvider;
20  import com.github.jpmsilva.jsystemd.StatusProvider;
21  import java.util.Collection;
22  import java.util.List;
23  import java.util.Locale;
24  import org.jspecify.annotations.Nullable;
25  import org.springframework.beans.factory.config.ConfigurableListableBeanFactory;
26  import org.springframework.jms.config.JmsListenerEndpointRegistry;
27  import org.springframework.jms.listener.DefaultMessageListenerContainer;
28  import org.springframework.jms.listener.MessageListenerContainer;
29  
30  /**
31   * Provides information regarding the JMS listener containers of the application, those of Spring: as a {@link StatusProvider}, how many of them run,
32   * out of those there are, and how many are recovering their connection to the broker, if any, such as {@code JMS listeners: 2/2 running, 1 recovering},
33   * and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy while any of them stopped by itself, if enabled.
34   *
35   * <p>The containers are those of the {@code @JmsListener} methods, held by the {@link JmsListenerEndpointRegistry}, and the
36   * {@link MessageListenerContainer} beans, found once all singleton beans are created, so that lazy ones are not created.
37   *
38   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
39   * unhealthy while any container that starts automatically ({@link MessageListenerContainer#isAutoStartup()}) does not run, or, for a
40   * {@link DefaultMessageListenerContainer}, runs without any consumer left, which leaves the application up, but no longer consuming, naming the containers
41   * in the health details, under {@code failedListenerContainers}. A {@link DefaultMessageListenerContainer} ends up in either state by itself once its
42   * back-off is exhausted, after failing to connect to the broker, or to set up its consumers, for as many attempts as it allows.
43   *
44   * <p>The containers do not tell why they are stopped, so a container that the application stops itself cannot be told apart from one that stopped by
45   * itself, and is reported as unhealthy too, including when the application context is stopped without being closed. Only enable the health in
46   * applications that leave the containers that start automatically running, or delay the unhealthy state for longer than they are stopped, with
47   * {@link #setUnhealthyDelay(java.time.Duration)}. A container that does not start automatically is left for the application to start and stop, and is
48   * always as expected.
49   *
50   * <p>By default, the back-off of a container is never exhausted: it keeps trying to connect to a broker that cannot be reached, every 5 seconds, and is
51   * then recovering, rather than stopped. That is left for the health indicators of Spring Boot Actuator.
52   *
53   * @author Joao Silva
54   */
55  public class JmsListenerStatusProvider extends ListenerContainerStatusProvider<JmsListenerEndpointRegistry, MessageListenerContainer> {
56  
57    /**
58     * Creates a new instance that finds the listener containers in the given bean factory.
59     *
60     * @param beanFactory the bean factory to find the listener endpoint registries and the listener containers in
61     * @param health <code>true</code> to report the application as unhealthy while any listener container stopped by itself
62     */
63    public JmsListenerStatusProvider(ConfigurableListableBeanFactory beanFactory, boolean health) {
64      super("JMS listeners", beanFactory, JmsListenerEndpointRegistry.class, MessageListenerContainer.class, health);
65    }
66  
67    @Override
68    Collection<String> ids(JmsListenerEndpointRegistry registry) {
69      return registry.getListenerContainerIds();
70    }
71  
72    @Override
73    @Nullable MessageListenerContainer container(JmsListenerEndpointRegistry registry, String id) {
74      return registry.getListenerContainer(id);
75    }
76  
77    @Override
78    boolean running(MessageListenerContainer container) {
79      return container.isRunning();
80    }
81  
82    @Override
83    boolean failed(MessageListenerContainer container) {
84      if (!container.isRunning()) {
85        return container.isAutoStartup();
86      }
87      // the consumers that are scheduled but paused are those that could not be scheduled again, as while the container is stopped
88      return container instanceof DefaultMessageListenerContainer consumers
89          && consumers.getScheduledConsumerCount() - consumers.getPausedTaskCount() < 1;
90    }
91  
92    @Override
93    List<String> details(Collection<MessageListenerContainer> containers) {
94      long recovering = containers.stream()
95          .filter(container -> container instanceof DefaultMessageListenerContainer recoverable && recoverable.isRecovering()).count();
96      return recovering == 0 ? List.of() : List.of(String.format(Locale.ROOT, "%d recovering", recovering));
97    }
98  }
