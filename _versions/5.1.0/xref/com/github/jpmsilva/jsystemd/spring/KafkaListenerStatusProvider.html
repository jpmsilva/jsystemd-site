17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.HealthProvider;
20  import com.github.jpmsilva.jsystemd.StatusProvider;
21  import java.util.ArrayList;
22  import java.util.Collection;
23  import java.util.List;
24  import java.util.Locale;
25  import org.apache.kafka.common.TopicPartition;
26  import org.jspecify.annotations.Nullable;
27  import org.springframework.beans.factory.config.ConfigurableListableBeanFactory;
28  import org.springframework.kafka.config.KafkaListenerEndpointRegistry;
29  import org.springframework.kafka.listener.MessageListenerContainer;
30  
31  /**
32   * Provides information regarding the Kafka listener containers of the application, those of Spring for Apache Kafka: as a {@link StatusProvider}, how
33   * many of them run, out of those there are, the partitions assigned to those that run, and how many are paused, if any, such as
34   * {@code Kafka listeners: 3/3 running, 12 partitions, 1 paused}, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as
35   * unhealthy while any of them stopped by itself, if enabled.
36   *
37   * <p>The containers are those of the {@code @KafkaListener} methods, held by the {@link KafkaListenerEndpointRegistry}, and the
38   * {@link MessageListenerContainer} beans, found once all singleton beans are created, so that lazy ones are not created. Only what the containers already
39   * know is reported, without asking the brokers, so that the lag of the consumers is not.
40   *
41   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
42   * unhealthy while any container is not in its expected state, as told by {@link MessageListenerContainer#isInExpectedState()}: it stopped by itself, after
43   * an error it does not recover from, such as when it is not authorized, when it is fenced, or when its error handler stops it, which leaves the
44   * application up, but no longer consuming, naming the containers in the health details, under {@code failedListenerContainers}. A container that was never
45   * started, or that the application stopped or paused, is as expected. As such a container can be started again without restarting the application, the
46   * unhealthy state can be delayed, with {@link #setUnhealthyDelay(java.time.Duration)}.
47   *
48   * <p>Brokers that cannot be reached do not stop the containers, which keep trying. That is left for the health indicators of Spring Boot Actuator.
49   *
50   * @author Joao Silva
51   */
52  public class KafkaListenerStatusProvider extends ListenerContainerStatusProvider<KafkaListenerEndpointRegistry, MessageListenerContainer> {
53  
54    /**
55     * Creates a new instance that finds the listener containers in the given bean factory.
56     *
57     * @param beanFactory the bean factory to find the listener endpoint registries and the listener containers in
58     * @param health <code>true</code> to report the application as unhealthy while any listener container stopped by itself
59     */
60    public KafkaListenerStatusProvider(ConfigurableListableBeanFactory beanFactory, boolean health) {
61      super("Kafka listeners", beanFactory, KafkaListenerEndpointRegistry.class, MessageListenerContainer.class, health);
62    }
63  
64    @Override
65    Collection<String> ids(KafkaListenerEndpointRegistry registry) {
66      return registry.getListenerContainerIds();
67    }
68  
69    @Override
70    @Nullable MessageListenerContainer container(KafkaListenerEndpointRegistry registry, String id) {
71      return registry.getListenerContainer(id);
72    }
73  
74    @Override
75    boolean running(MessageListenerContainer container) {
76      return container.isRunning();
77    }
78  
79    @Override
80    boolean failed(MessageListenerContainer container) {
81      return !container.isInExpectedState();
82    }
83  
84    @Override
85    List<String> details(Collection<MessageListenerContainer> containers) {
86      int partitions = 0;
87      boolean assignable = false;
88      int paused = 0;
89      for (MessageListenerContainer container : containers) {
90        int assigned = partitions(container);
91        if (assigned >= 0) {
92          assignable = true;
93          partitions += assigned;
94        }
95        if (paused(container)) {
96          paused++;
97        }
98      }
99      List<String> details = new ArrayList<>();
100     if (assignable) {
101       details.add(String.format(Locale.ROOT, "%d %s", partitions, partitions == 1 ? "partition" : "partitions"));
102     }
103     if (paused > 0) {
104       details.add(String.format(Locale.ROOT, "%d paused", paused));
105     }
106     return details;
107   }
108 
109   /**
110    * Returns the number of partitions assigned to the given container, or -1 when it does not tell, such as the containers of share groups, which are
111    * not assigned any.
112    */
113   private static int partitions(MessageListenerContainer container) {
114     try {
115       // null until partitions are assigned, and a container that is stopped still tells the ones it had
116       Collection<TopicPartition> assigned = container.getAssignedPartitions();
117       return assigned == null || !container.isRunning() ? 0 : assigned.size();
118     } catch (UnsupportedOperationException e) {
119       return -1;
120     }
121   }
122 
123   /**
124    * Tells if the given container is paused, which only a running container is, when it can be paused at all.
125    */
126   private static boolean paused(MessageListenerContainer container) {
127     try {
128       return container.isRunning() && container.isContainerPaused();
129     } catch (UnsupportedOperationException e) {
130       return false;
131     }
132   }
133 }
