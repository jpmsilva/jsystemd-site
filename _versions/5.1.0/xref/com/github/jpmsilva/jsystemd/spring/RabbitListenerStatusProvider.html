18  
19  import com.github.jpmsilva.jsystemd.HealthProvider;
20  import com.github.jpmsilva.jsystemd.StatusProvider;
21  import java.util.Collection;
22  import java.util.Collections;
23  import java.util.IdentityHashMap;
24  import java.util.List;
25  import java.util.Locale;
26  import java.util.Set;
27  import org.jspecify.annotations.Nullable;
28  import org.springframework.amqp.rabbit.listener.ListenerContainerConsumerFailedEvent;
29  import org.springframework.amqp.rabbit.listener.MessageListenerContainer;
30  import org.springframework.amqp.rabbit.listener.RabbitListenerEndpointRegistry;
31  import org.springframework.amqp.rabbit.listener.SimpleMessageListenerContainer;
32  import org.springframework.beans.factory.config.ConfigurableListableBeanFactory;
33  import org.springframework.context.ApplicationListener;
34  
35  /**
36   * Provides information regarding the RabbitMQ listener containers of the application, those of Spring AMQP: as a {@link StatusProvider}, how many of
37   * them run, out of those there are, and the active consumers of the {@link SimpleMessageListenerContainer} ones, such as
38   * {@code RabbitMQ listeners: 2/2 running, 4 consumers}, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy
39   * while any of them stopped by itself, if enabled.
40   *
41   * <p>The containers are those of the {@code @RabbitListener} methods, held by the {@link RabbitListenerEndpointRegistry}, and the
42   * {@link MessageListenerContainer} beans, found once all singleton beans are created, so that lazy ones are not created. The active consumers drop to
43   * none while the broker cannot be reached, as the containers keep running, trying to start their consumers again.
44   *
45   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
46   * unhealthy while any container stopped by itself: one of its consumers failed with a fatal error, such as when its queues are missing, when it is not
47   * authenticated, or when its listener throws an {@link Error}, which stops the container, and it has not been started again, which leaves the
48   * application up, but no longer consuming, naming the containers in the health details, under {@code failedListenerContainers}. The containers do not
49   * tell why they are stopped, so the fatal errors are learned from the {@link ListenerContainerConsumerFailedEvent} events they publish: this provider
50   * must be a bean of the application context of the containers to receive them. A container that was never started, or that the application stopped, is
51   * as expected. As such a container can be started again without restarting the application, the unhealthy state can be delayed, with
52   * {@link #setUnhealthyDelay(java.time.Duration)}.
53   *
54   * <p>Not every container that stops by itself publishes such an event: a {@link org.springframework.amqp.rabbit.listener.DirectMessageListenerContainer}
55   * that gives up connecting to the broker, once its recovery back-off is exhausted, which it never is by default, does not, and is not reported as
56   * unhealthy. A broker that cannot be reached does not stop the containers, which keep trying by default. That is left for the health indicators of Spring
57   * Boot Actuator.
58   *
59   * @author Joao Silva
60   */
61  public class RabbitListenerStatusProvider extends ListenerContainerStatusProvider<RabbitListenerEndpointRegistry, MessageListenerContainer>
62      implements ApplicationListener<ListenerContainerConsumerFailedEvent> {
63  
64    // the containers that had a consumer fail with a fatal error, until they are found running again
65    private final Set<Object> failures = Collections.synchronizedSet(Collections.newSetFromMap(new IdentityHashMap<>()));
66  
67    /**
68     * Creates a new instance that finds the listener containers in the given bean factory.
69     *
70     * @param beanFactory the bean factory to find the listener endpoint registries and the listener containers in
71     * @param health <code>true</code> to report the application as unhealthy while any listener container stopped by itself
72     */
73    public RabbitListenerStatusProvider(ConfigurableListableBeanFactory beanFactory, boolean health) {
74      super("RabbitMQ listeners", beanFactory, RabbitListenerEndpointRegistry.class, MessageListenerContainer.class, health);
75    }
76  
77    /**
78     * Takes note of the containers that have a consumer fail with a fatal error, which stops them.
79     *
80     * @param event the failure of a consumer of a listener container
81     */
82    @Override
83    public void onApplicationEvent(ListenerContainerConsumerFailedEvent event) {
84      // a consumer that is interrupted is one that takes too long to stop, in a container that is being stopped, rather than one that failed
85      if (event.isFatal() && !(event.getThrowable() instanceof InterruptedException)) {
86        failures.add(event.getSource());
87      }
88    }
89  
90    @Override
91    Collection<String> ids(RabbitListenerEndpointRegistry registry) {
92      return registry.getListenerContainerIds();
93    }
94  
95    @Override
96    @Nullable MessageListenerContainer container(RabbitListenerEndpointRegistry registry, String id) {
97      return registry.getListenerContainer(id);
98    }
99  
100   @Override
101   boolean running(MessageListenerContainer container) {
102     return container.isRunning();
103   }
104 
105   @Override
106   boolean failed(MessageListenerContainer container) {
107     if (!failures.contains(container)) {
108       return false;
109     }
110     if (container.isRunning()) {
111       // started again since
112       failures.remove(container);
113       return false;
114     }
115     return true;
116   }
117 
118   @Override
119   List<String> details(Collection<MessageListenerContainer> containers) {
120     // the other containers do not tell how many consumers they have
121     List<SimpleMessageListenerContainer> simple = containers.stream()
122         .filter(SimpleMessageListenerContainer.class::isInstance).map(SimpleMessageListenerContainer.class::cast).toList();
123     if (simple.isEmpty()) {
124       return List.of();
125     }
126     int consumers = simple.stream().mapToInt(SimpleMessageListenerContainer::getActiveConsumerCount).sum();
127     return List.of(String.format(Locale.ROOT, "%d %s", consumers, consumers == 1 ? "consumer" : "consumers"));
128   }
129 }
