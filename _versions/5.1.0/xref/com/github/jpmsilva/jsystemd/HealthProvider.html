18  package com.github.jpmsilva.jsystemd;
19  
20  import static java.util.Objects.requireNonNull;
21  
22  import java.util.Collections;
23  import java.util.LinkedHashMap;
24  import java.util.Map;
25  
26  /**
27   * A health provider can be used to control the watchdog heartbeat. An unhealthy state does not trigger the heartbeat, so that systemd restarts the
28   * application once <code>WatchdogSec</code> elapses. An unhealthy state that the application cannot recover from also triggers the watchdog right away.
29   *
30   * <p>Health providers only report the health: they must not drive the lifecycle of the application with {@link Systemd#ready()},
31   * {@link Systemd#stopping()} or {@link Systemd#close()}, as they are called from the periodic watchdog heartbeats, which {@link Systemd#close()} waits for.
32   *
33   * @author Christian Lorenz
34   */
35  public interface HealthProvider {
36  
37    /**
38     * Determines if an application is healthy.
39     *
40     * @return representation of the application {@link Health} state
41     */
42    Health health();
43  
44    /**
45     * The health state of an application. Immutable: the details are copied, keeping their order, and cannot be modified.
46     *
47     * <p>An unhealthy state suppresses the watchdog heartbeat for as long as it lasts, so that systemd restarts the application once
48     * <code>WatchdogSec</code> elapses, unless the application is healthy again by then. An unrecoverable one, which never resolves by itself, such as a
49     * deadlock, also triggers the watchdog right away, with {@link Systemd#triggerWatchdog()}, as waiting would only delay the recovery.
50     *
51     * @param healthy <code>true</code> if this object represents a healthy state, <code>false</code> otherwise
52     * @param details additional health details for each application component
53     * @param unrecoverable <code>true</code> if this object represents an unhealthy state that the application cannot recover from
54     */
55    record Health(boolean healthy, Map<String, Object> details, boolean unrecoverable) {
56  
57      /**
58       * A health object that represents a healthy application state.
59       */
60      public static final Health HEALTHY = new Health(true, Collections.emptyMap());
61  
62      /**
63       * Provides for a health representation (healthy/unhealthy/unrecoverable), with health details for each component.
64       *
65       * @param healthy <code>true</code> if this object represents a healthy state, <code>false</code> otherwise
66       * @param details additional health details for each application component
67       * @param unrecoverable <code>true</code> if this object represents an unhealthy state that the application cannot recover from
68       * @throws NullPointerException if details is null
69       * @throws IllegalArgumentException if a healthy state is unrecoverable
70       */
71      public Health {
72        if (healthy && unrecoverable) {
73          throw new IllegalArgumentException("A healthy state cannot be unrecoverable");
74        }
75        details = Collections.unmodifiableMap(new LinkedHashMap<>(requireNonNull(details, "Details must not be null")));
76      }
77  
78      /**
79       * Provides for a health representation (healthy/unhealthy), with health details for each component. An unhealthy state is one the application may
80       * recover from: see {@link #unrecoverable(Map)} for one it cannot.
81       *
82       * @param healthy <code>true</code> if this object represents a healthy state, <code>false</code> otherwise
83       * @param details additional health details for each application component
84       * @throws NullPointerException if details is null
85       */
86      public Health(boolean healthy, Map<String, Object> details) {
87        this(healthy, details, false);
88      }
89  
90      /**
91       * Provides for an unhealthy state that the application cannot recover from, with health details for each component.
92       *
93       * @param details additional health details for each application component
94       * @return an unhealthy state that the application cannot recover from
95       * @throws NullPointerException if details is null
96       */
97      public static Health unrecoverable(Map<String, Object> details) {
98        return new Health(false, details, true);
99      }
100   }
101 }
