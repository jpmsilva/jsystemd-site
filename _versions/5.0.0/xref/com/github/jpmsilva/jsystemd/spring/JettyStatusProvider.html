17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import com.github.jpmsilva.jsystemd.StatusProvider;
20  import java.util.ArrayList;
21  import java.util.List;
22  import java.util.Locale;
23  import java.util.Map;
24  import java.util.TreeMap;
25  import java.util.concurrent.CopyOnWriteArrayList;
26  import org.eclipse.jetty.server.AbstractConnector;
27  import org.eclipse.jetty.server.Connector;
28  import org.eclipse.jetty.server.NetworkConnectionLimit;
29  import org.eclipse.jetty.server.NetworkConnector;
30  import org.eclipse.jetty.server.Server;
31  import org.eclipse.jetty.util.thread.QueuedThreadPool;
32  import org.jspecify.annotations.Nullable;
33  import org.springframework.boot.jetty.JettyWebServer;
34  import org.springframework.boot.web.server.context.WebServerInitializedEvent;
35  import org.springframework.context.ApplicationListener;
36  
37  /**
38   * Implementation of {@link StatusProvider} that provides information regarding the embedded Jetty servers of the application.
39   *
40   * <p>For each server, named after the port it listens on, it reports the open connections, out of the maximum the server accepts when limited, and, when
41   * the requests are handled by a thread pool rather than by virtual threads, the threads running requests out of the threads in the pool, such as
42   * {@code jetty-8080: 12/1000 connections, 3/10 threads}. The threads Jetty keeps for itself, to accept connections and wait for their data, are left out of
43   * the threads running requests. A server that is not started is left out.
44   *
45   * <p>The servers are found as they start, from their {@link WebServerInitializedEvent}: the server of the application, as well as the server of the
46   * management endpoints when they use a port of their own. Nothing is reported until then.
47   *
48   * @author Joao Silva
49   */
50  public class JettyStatusProvider implements StatusProvider, ApplicationListener<WebServerInitializedEvent> {
51  
52    // added by the event listener, read by the status updates
53    private final CopyOnWriteArrayList<Server> servers = new CopyOnWriteArrayList<>();
54  
55    /**
56     * Create a new JettyStatusProvider.
57     */
58    public JettyStatusProvider() {
59    }
60  
61    /**
62     * Adds the server of the event to the servers that are reported, if it is a Jetty server.
63     *
64     * @param event the event of a web server that started
65     */
66    @Override
67    public void onApplicationEvent(WebServerInitializedEvent event) {
68      if (event.getWebServer() instanceof JettyWebServer server) {
69        servers.addIfAbsent(server.getServer());
70      }
71    }
72  
73    @Override
74    public String status() {
75      // sorted by name, so that the order does not depend on the order in which the servers start
76      Map<String, String> statuses = new TreeMap<>();
77      for (Server server : servers) {
78        String name = name(server);
79        if (name != null) {
80          statuses.put(name, name + ": " + String.join(", ", details(server)));
81        }
82      }
83      return String.join(", ", statuses.values());
84    }
85  
86    /**
87     * Returns the name of the server, after the ports its connectors listen on, or <code>null</code> if none is listening.
88     */
89    private static @Nullable String name(Server server) {
90      List<String> ports = new ArrayList<>();
91      for (Connector connector : server.getConnectors()) {
92        // -1 while the connector is not started
93        if (connector instanceof NetworkConnector network && network.getLocalPort() > 0) {
94          ports.add(String.valueOf(network.getLocalPort()));
95        }
96      }
97      return ports.isEmpty() ? null : "jetty-" + String.join("+", ports);
98    }
99  
100   private static List<String> details(Server server) {
101     List<String> details = new ArrayList<>();
102     int connections = 0;
103     for (Connector connector : server.getConnectors()) {
104       if (connector instanceof AbstractConnector abstractConnector) {
105         connections += abstractConnector.getConnectedEndPoints().size();
106       }
107     }
108     // the maximum is only set with server.jetty.max-connections, which limits the connections of all the connectors of the server together
109     NetworkConnectionLimit limit = server.getBean(NetworkConnectionLimit.class);
110     details.add(limit != null ? String.format(Locale.ROOT, "%d/%d connections", connections, limit.getMaxNetworkConnectionCount())
111         : String.format(Locale.ROOT, "%d connections", connections));
112     // virtual threads are not pooled, so there is nothing to report on them; the threads leased by Jetty, the acceptors and selectors, never run requests
113     if (server.getThreadPool() instanceof QueuedThreadPool pool) {
114       details.add(String.format(Locale.ROOT, "%d/%d threads", pool.getUtilizedThreads(), pool.getThreads() - pool.getLeasedThreads()));
115     }
116     return details;
117   }
118 }
