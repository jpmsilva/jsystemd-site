109   /**
110    * Event listener for the {@link AvailabilityChangeEvent} event to report to systemd that the service is ready.
111    *
112    * <p>The application is considered ready when the event is a {@link ReadinessState} with the state {@link ReadinessState#ACCEPTING_TRAFFIC}, unless
113    * configured to be ready once it has started instead, with <code>systemd.ready-event=application-started</code>, which
114    * {@link SystemdSpringApplicationRunListener} reports.
115    *
116    * @param event the {@link AvailabilityChangeEvent} received
117    */
118   @EventListener
119   public void started(@SuppressWarnings("unused") AvailabilityChangeEvent<ReadinessState> event) {
120     if (properties.getReadyEvent() == SystemdProperties.ReadyEvent.ACCEPTING_TRAFFIC && event.getState() == ReadinessState.ACCEPTING_TRAFFIC) {
121       Systemd.instance().ready();
122     }
123   }

389   /**
390    * Configuration class for systemd integration when using Spring Boot Actuator, providing a {@link HealthProvider} based on its health contributors.
391    */
392   @Configuration(proxyBeanMethods = false)
393   @ConditionalOnSystemd
394   @ConditionalOnClass(HealthIndicator.class)
395   @ConditionalOnProperty(prefix = SystemdProperties.STATUS_PROVIDERS_PREFIX, name = "actuator-health.enabled", matchIfMissing = true)
396   public static class SystemdActuatorHealthConfiguration {
397 
398     SystemdActuatorHealthConfiguration() {
399     }
400 
401     @Bean
402     @Order(-8000)
403     @ConditionalOnMissingBean(ActuatorHealthStatusProvider.class)
404     ActuatorHealthStatusProvider systemdActuatorHealthStatusProvider(ObjectProvider<HealthContributorRegistry> registryProvider,
405         ObjectProvider<ReactiveHealthContributorRegistry> reactiveRegistryProvider, ObjectProvider<HealthEndpointGroups> groupsProvider,
406         SystemdProperties systemdProperties) {
407       requireNonNull(registryProvider, "Registry provider must not be null");
408       requireNonNull(reactiveRegistryProvider, "Reactive registry provider must not be null");
409       requireNonNull(systemdProperties, "Properties must not be null");
410       SystemdProperties.ActuatorHealth properties = systemdProperties.getStatusProviders().getActuatorHealth();
411 
412       // the same contributors, with the same names, as the health endpoint, which also adapts the reactive ones
413       List<HealthContributors> contributors = new ArrayList<>();
414       HealthContributorRegistry registry = registryProvider.getIfAvailable();
415       if (registry == null) {
416         logger.warn("There is no health contributor registry, so the Spring Boot Actuator health is always reported as healthy; check that the health "
417             + "contributor registry autoconfiguration of Spring Boot is not excluded");
418       } else {
419         contributors.add(registry);
420       }
421       reactiveRegistryProvider.ifAvailable(reactiveRegistry -> contributors.add(reactiveRegistry.asHealthContributors()));
422       Set<Status> unhealthyStatusCodes = properties.getUnhealthyStatusCodes().stream().map(Status::new).collect(Collectors.toSet());
423       if (properties.isHealth() && unhealthyStatusCodes.isEmpty() && Systemd.watchdogTimeout().isPresent() && !systemdProperties.isWatchdogSuppressWarning()) {
424         logger.warn("The systemd watchdog is enabled but no health status codes are considered as unhealthy, so the watchdog heartbeat will never be "
425             + "suppressed; set '{}.unhealthy-status-codes' with the status codes to consider as unhealthy, or suppress this warning with "
426             + "'{}.watchdog-suppress-warning=true'", SystemdProperties.ActuatorHealth.PREFIX, SystemdProperties.PREFIX);
427       }
428       ActuatorHealthStatusProvider provider = new ActuatorHealthStatusProvider(HealthContributors.of(contributors.toArray(HealthContributors[]::new)),
429           unhealthyStatusCodes, member(properties.getGroup(), groupsProvider), properties.isHealth());
430       provider.setUnhealthyDelay(properties.getUnhealthyDelay());
431       return provider;
432     }
433 
434     /**
435      * Returns the membership test of the health group with the given name, the one of the health endpoint, or one that takes all the health indicators into
436      * account without a group.
437      *
438      * @throws InvalidConfigurationPropertyValueException if there is no health group with the given name
439      */
440     private static Predicate<String> member(@Nullable String group, ObjectProvider<HealthEndpointGroups> groupsProvider) {
441       if (group == null) {
442         return name -> true;
443       }
444       String property = SystemdProperties.ActuatorHealth.PREFIX + ".group";
445       HealthEndpointGroups groups = groupsProvider.getIfAvailable();
446       if (groups == null) {
447         throw new InvalidConfigurationPropertyValueException(property, group, "There are no health groups, as the health endpoint of Spring Boot "
448             + "Actuator is not available: check that it is not disabled, nor its autoconfiguration excluded");
449       }
450       HealthEndpointGroup healthGroup = groups.get(group);
451       if (healthGroup == null) {
452         throw new InvalidConfigurationPropertyValueException(property, group, "There is no such health group; the health groups are " + groups.getNames()
453             + " (the liveness and readiness groups are not set up with management.endpoint.health.probes.enabled=false)");
454       }
455       return healthGroup::isMember;
456     }
457   }

459   /**
460    * Makes the systemd status include the status of the application, once all beans are created, so that the {@link StatusProvider} beans defined by
461    * the application are also found. The beans are added to the status providers of systemd, after the ones already registered (such as the startup status
462    * providers), in the order given by the <code>@Order</code> of their bean methods or classes. Beans already registered are not added again. While there
463    * is no status provider, no status is sent.
464    *
465    * <p>The built-in status providers, such as the {@link HeapStatusProvider}, are added while enabled, with the
466    * <code>systemd.status-providers.&lt;key&gt;.enabled</code> properties, all enabled by default except for the {@link GarbageCollectionStatusProvider}
467    * and the {@link CpuStatusProvider}, which are opted into.
468    *
469    * @param statuses the status providers of the application
470    * @return the callback that adds the status providers to the systemd status
471    */
472   @Bean
473   SmartInitializingSingleton systemdStatusProviders(ObjectProvider<StatusProvider> statuses) {
474     return () -> {
475       Systemd systemd = Systemd.instance();
476       List<StatusProvider> registered = systemd.getStatusProviders();
477       systemd.addStatusProviders(statuses.orderedStream()
478           .filter(provider -> !registered.contains(provider))
479           .toList());
480     };
481   }

483   /**
484    * Makes the systemd watchdog take the application health into account, once all beans are created, so that the {@link HealthProvider} beans defined by
485    * the application are also found. The beans are added to the health providers of systemd, after the ones already registered, in the order given by the
486    * <code>@Order</code> of their bean methods or classes. Beans already registered are not added again. While there is no health provider, the watchdog
487    * heartbeat is always sent.
488    *
489    * <p>The built-in status providers that are also health providers, such as the {@link ThreadStatusProvider}, the {@link FileDescriptorStatusProvider},
490    * the {@link HikariStatusProvider}, the {@link AvailabilityStatusProvider} and the {@link ActuatorHealthStatusProvider}, are added while enabled. Their
491    * health conditions are controlled by their own properties, all disabled by default, so that they are always healthy unless opted into.
492    *
493    * @param healthProviders the health providers of the application
494    * @return the callback that adds the health providers to the watchdog
495    */
496   @Bean
497   SmartInitializingSingleton systemdHealthProviders(ObjectProvider<HealthProvider> healthProviders) {
498     return () -> {
499       Systemd systemd = Systemd.instance();
500       List<HealthProvider> registered = systemd.getHealthProviders();
501       systemd.addHealthProviders(healthProviders.orderedStream()
502           .filter(provider -> !registered.contains(provider))
503           .toList());
504     };
505   }
