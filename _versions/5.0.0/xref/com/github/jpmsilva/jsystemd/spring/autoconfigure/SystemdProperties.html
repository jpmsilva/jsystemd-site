58  /**
59   * Properties of the integration with systemd when using Spring Boot, all under <code>systemd</code>.
60   *
61   * <ul>
62   *   <li><code>systemd.ready-event</code>, when systemd is told that the application is ready;</li>
63   *   <li><code>systemd.status-update-period</code>, how often the status is sent to systemd;</li>
64   *   <li><code>systemd.watchdog-suppress-warning</code>, whether to log the warnings about the health signalled to the systemd watchdog;</li>
65   *   <li><code>systemd.status-providers.&lt;key&gt;.*</code>, each built-in {@link StatusProvider}: whether it is enabled, all of them by default
66   *   except for the {@link GarbageCollectionStatusProvider} and the {@link CpuStatusProvider}, which are opted into, to keep the status short, and the
67   *   other settings of those that can be configured.</li>
68   * </ul>
69   *
70   * <p>Invalid values are rejected when the properties are bound, failing the startup of the application with an
71   * {@link InvalidConfigurationPropertyValueException} that names the property.
72   *
73   * @author Joao Silva
74   * @author Christian Lorenz
75   */

561   /**
562    * The {@link ThreadStatusProvider}, which provides information regarding the live threads, and detects deadlocks.
563    *
564    * <p>The {@link ThreadStatusProvider} is always a health provider of the watchdog, while enabled, and these properties control its health conditions.
565    * Each health condition is opted into on its own, as it lets systemd restart the application: without any, the application is always healthy.
566    *
567    * <p>Both deadlock detection and deadlock health are disabled by default: detecting deadlocks shows them in the systemd status, and deadlock health
568    * makes the application unhealthy while they last. Deadlock health requires detecting deadlocks: enabling <code>deadlock-health</code> while
569    * <code>deadlock-detection</code> is disabled is an error, which fails the startup of the application.
570    */
571   public static class Threads extends Toggle {
572 
573     private static final String PREFIX = STATUS_PROVIDERS_PREFIX + ".threads";
574 
575     /**
576      * Detect deadlocked threads, and report them in the systemd status.
577      */
578     private boolean deadlockDetection;
579 
580     /**
581      * Suppress the watchdog heartbeat while any thread is deadlocked, so that systemd restarts the application. Requires deadlock detection: enabling it
582      * while deadlock detection is disabled is an error.
583      */
584     private boolean deadlockHealth;
585 
586     Threads() {
587     }
588 
589     /**
590      * Check if deadlocked threads are detected.
591      *
592      * @return <code>true</code> if deadlocked threads are detected
593      */
594     public boolean isDeadlockDetection() {
595       return deadlockDetection;
596     }
597 
598     /**
599      * Enable or disable the detection of deadlocked threads.
600      *
601      * @param deadlockDetection <code>true</code> to detect deadlocked threads
602      */
603     public void setDeadlockDetection(boolean deadlockDetection) {
604       this.deadlockDetection = deadlockDetection;
605     }
606 
607     /**
608      * Check if the watchdog heartbeat is suppressed while any thread is deadlocked.
609      *
610      * @return <code>true</code> if deadlocked threads make the application unhealthy
611      */
612     public boolean isDeadlockHealth() {
613       return deadlockHealth;
614     }
615 
616     /**
617      * Enable or disable suppressing the watchdog heartbeat while any thread is deadlocked. Requires deadlock detection.
618      *
619      * @param deadlockHealth <code>true</code> to make the application unhealthy while any thread is deadlocked
620      */
621     public void setDeadlockHealth(boolean deadlockHealth) {
622       this.deadlockHealth = deadlockHealth;
623     }
624 
625     // rejects signalling deadlocks without detecting them, which would otherwise silently never signal anything
626     void validate() {
627       if (deadlockHealth && !deadlockDetection) {
628         throw new InvalidConfigurationPropertyValueException(PREFIX + ".deadlock-health", true,
629             "Signalling deadlocks requires detecting them: set '" + PREFIX + ".deadlock-detection=true', or remove '" + PREFIX + ".deadlock-health'");
630       }
631     }
632   }

634   /**
635    * The {@link FileDescriptorStatusProvider}, which provides information regarding the open file descriptors.
636    *
637    * <p>The {@link FileDescriptorStatusProvider} is always a health provider of the watchdog, while enabled, and these properties control its health: it
638    * is opted into, as it lets systemd restart the application, so the application is always healthy unless enabled.
639    */
640   public static class FileDescriptors extends Toggle {
641 
642     private static final String PREFIX = STATUS_PROVIDERS_PREFIX + ".file-descriptors";
643 
644     /**
645      * Suppress the watchdog heartbeat while the open file descriptors are at or above the threshold, so that systemd restarts the application.
646      */
647     private boolean health;
648 
649     /**
650      * Share of the limit of file descriptors from which the application is unhealthy, greater than 0 and at most 1, such as <code>0.9</code> for 90%.
651      * Only applies with health enabled.
652      */
653     private double threshold = 0.9;
654 
655     /**
656      * Delay suppressing the watchdog heartbeat until the open file descriptors have been at or above the threshold for this long, such as
657      * <code>30s</code>, so that a short burst of connections does not restart the application. Values without a unit are in milliseconds. Not set by
658      * default, which disables the delay. Only applies with health enabled.
659      */
660     @Nullable
661     @DurationUnit(ChronoUnit.MILLIS)
662     private Duration unhealthyDelay;
663 
664     FileDescriptors() {
665     }
666 
667     /**
668      * Check if the watchdog heartbeat is suppressed while the open file descriptors are at or above the threshold.
669      *
670      * @return <code>true</code> if the file descriptors about to run out make the application unhealthy
671      */
672     public boolean isHealth() {
673       return health;
674     }
675 
676     /**
677      * Enable or disable suppressing the watchdog heartbeat while the open file descriptors are at or above the threshold.
678      *
679      * @param health <code>true</code> to make the application unhealthy when the file descriptors are about to run out
680      */
681     public void setHealth(boolean health) {
682       this.health = health;
683     }
684 
685     /**
686      * Check the share of the limit of file descriptors from which the application is unhealthy.
687      *
688      * @return the share of the limit, greater than 0 and at most 1
689      */
690     public double getThreshold() {
691       return threshold;
692     }
693 
694     /**
695      * Set the share of the limit of file descriptors from which the application is unhealthy.
696      *
697      * @param threshold the share of the limit, greater than 0 and at most 1
698      */
699     public void setThreshold(double threshold) {
700       this.threshold = threshold;
701     }
702 
703     /**
704      * Check the delay before suppressing the watchdog heartbeat while the open file descriptors are at or above the threshold.
705      *
706      * @return the delay, or <code>null</code> if there is no delay
707      */
708     public @Nullable Duration getUnhealthyDelay() {
709       return unhealthyDelay;
710     }
711 
712     /**
713      * Set the delay before suppressing the watchdog heartbeat while the open file descriptors are at or above the threshold.
714      *
715      * @param unhealthyDelay the delay, or <code>null</code> to disable the delay
716      */
717     public void setUnhealthyDelay(@Nullable Duration unhealthyDelay) {
718       this.unhealthyDelay = unhealthyDelay;
719     }
720 
721     // rejects thresholds out of range, which would otherwise make the application always or never unhealthy
722     void validate() {
723       if (!(threshold > 0 && threshold <= 1)) {
724         throw new InvalidConfigurationPropertyValueException(PREFIX + ".threshold", threshold, "The threshold must be greater than 0 and at most 1");
725       }
726       DurationUtilities.requireNotNegative(PREFIX + ".unhealthy-delay", unhealthyDelay);
727     }
728   }

730   /**
731    * The {@link AvailabilityStatusProvider}, which provides the liveness and readiness of the application.
732    *
733    * <p>The {@link AvailabilityStatusProvider} is always a health provider of the watchdog, while enabled, and these properties control its health: it is
