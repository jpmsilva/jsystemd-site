17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import static com.github.jpmsilva.jsystemd.spring.BeanFactoryUtilities.createdBeans;
20  import static java.util.Objects.requireNonNull;
21  
22  import com.github.jpmsilva.jsystemd.ExecutorStatusProvider;
23  import com.github.jpmsilva.jsystemd.StatusProvider;
24  import java.util.ArrayList;
25  import java.util.List;
26  import java.util.Map;
27  import java.util.TreeMap;
28  import java.util.concurrent.ThreadPoolExecutor;
29  import java.util.function.Supplier;
30  import org.jspecify.annotations.Nullable;
31  import org.springframework.beans.factory.SmartInitializingSingleton;
32  import org.springframework.beans.factory.config.ConfigurableListableBeanFactory;
33  import org.springframework.scheduling.concurrent.ThreadPoolTaskExecutor;
34  import org.springframework.scheduling.concurrent.ThreadPoolTaskScheduler;
35  
36  /**
37   * Implementation of {@link StatusProvider} that provides information regarding the thread pools of the task executors and schedulers of the
38   * application, such as the ones Spring Boot sets up for {@code @Async} and {@code @Scheduled} methods, each reported by an
39   * {@link ExecutorStatusProvider} named after its bean, such as {@code applicationTaskExecutor: 2/8 threads, 5 queued, taskScheduler: 1/1 threads, 3 scheduled}.
40   *
41   * <p>The task executors are the {@link ThreadPoolTaskExecutor} and {@link ThreadPoolTaskScheduler} beans, found once all singleton beans are created, so
42   * that lazy ones are not created. Those that run tasks on virtual threads, as Spring Boot sets them up with
43   * {@code spring.threads.virtual.enabled=true}, have no thread pool, and are left out.
44   *
45   * @author Joao Silva
46   */
47  public class TaskExecutorStatusProvider implements StatusProvider, SmartInitializingSingleton {
48  
49    private final ConfigurableListableBeanFactory beanFactory;
50  
51    // set once all singleton beans are created, read by the status updates
52    private volatile List<ExecutorStatusProvider> executors = List.of();
53  
54    /**
55     * Creates a new instance that finds the task executors in the given bean factory.
56     *
57     * @param beanFactory the bean factory to find the task executors in
58     */
59    public TaskExecutorStatusProvider(ConfigurableListableBeanFactory beanFactory) {
60      this.beanFactory = requireNonNull(beanFactory, "Bean factory must not be null");
61    }
62  
63    /**
64     * Finds the task executors among the singleton beans that are already created.
65     */
66    @Override
67    public void afterSingletonsInstantiated() {
68      // sorted by name, so that the order does not depend on the order in which the beans are defined
69      Map<String, ExecutorStatusProvider> found = new TreeMap<>();
70      createdBeans(beanFactory, ThreadPoolTaskExecutor.class).forEach((name, executor) ->
71          found.put(name, new ExecutorStatusProvider(name, initialized(executor::getThreadPoolExecutor))));
72      createdBeans(beanFactory, ThreadPoolTaskScheduler.class).forEach((name, scheduler) ->
73          found.put(name, new ExecutorStatusProvider(name, initialized(scheduler::getScheduledThreadPoolExecutor))));
74      executors = List.copyOf(found.values());
75    }
76  
77    /**
78     * Returns the thread pool given by the supplier, or <code>null</code> while the task executor is not initialized.
79     */
80    private static Supplier<@Nullable ThreadPoolExecutor> initialized(Supplier<? extends ThreadPoolExecutor> executor) {
81      return () -> {
82        try {
83          return executor.get();
84        } catch (IllegalStateException e) {
85          return null;
86        }
87      };
88    }
89  
90    @Override
91    public String status() {
92      List<String> statuses = new ArrayList<>();
93      for (ExecutorStatusProvider executor : executors) {
94        String status = executor.status();
95        if (!status.isEmpty()) {
96          statuses.add(status);
97        }
98      }
99      return String.join(", ", statuses);
100   }
101 }
