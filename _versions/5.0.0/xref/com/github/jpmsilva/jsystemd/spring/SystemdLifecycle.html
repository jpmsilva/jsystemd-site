17  package com.github.jpmsilva.jsystemd.spring;
18  
19  import static java.util.Objects.requireNonNull;
20  
21  import com.github.jpmsilva.jsystemd.Systemd;
22  import org.springframework.context.ApplicationContext;
23  import org.springframework.context.ApplicationListener;
24  import org.springframework.context.event.ContextClosedEvent;
25  
26  /**
27   * Sends the stopping signal to systemd when the application context starts closing, before its life cycle beans stop, such as the embedded web server,
28   * which then waits for the active requests.
29   *
30   * <p>Stopping the context without closing it does not send it, as when the context is stopped or paused and started again afterwards, such as by the pause
31   * endpoint of Spring Cloud, or for a CRaC checkpoint: the application keeps running, and systemd keeps enforcing the watchdog.
32   *
33   * <p>Register it as a bean of the application context it is given, for that context to publish its events to it.
34   *
35   * @author Joao Silva
36   * @see Systemd#stopping()
37   */
38  public class SystemdLifecycle implements ApplicationListener<ContextClosedEvent> {
39  
40    private final ApplicationContext context;
41  
42    /**
43     * Creates a new instance, which sends the stopping signal with {@link Systemd#instance()} when the given context starts closing.
44     *
45     * @param context the application context of the application, which this instance must be a bean of, rather than a parent or child context
46     */
47    public SystemdLifecycle(ApplicationContext context) {
48      this.context = requireNonNull(context, "Application context must not be null");
49    }
50  
51    /**
52     * Sends the stopping signal, when the event is about its own application context: the events of child contexts, such as the one of the management
53     * endpoints when they use a port of their own, are also published to their parents.
54     *
55     * <p>Spring publishes the event first when closing a context, before it stops the life cycle beans and destroys the beans, despite the wording of the
56     * documentation of {@link ContextClosedEvent}: the application is still running when the event is received.
57     *
58     * @param event the event of a context that started closing
59     */
60    @Override
61    public void onApplicationEvent(ContextClosedEvent event) {
62      if (event.getApplicationContext() == context) {
63        Systemd.instance().stopping();
64      }
65    }
66  }
