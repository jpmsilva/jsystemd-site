41  /**
42   * Provides information regarding the HikariCP connection pools of the application: as a {@link StatusProvider}, for each pool, named after its
43   * data source bean, the connections in use out of the most the pool opens, and the threads waiting for a connection, such as
44   * {@code dataSource: 7/10 connections, 3 waiting}, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy while
45   * a pool is starved, if enabled.
46   *
47   * <p>The pools are those of the {@link DataSource} beans that are, or wrap, a {@link HikariDataSource}, found once all singleton beans are created, so
48   * that lazy ones are not created. A data source whose pool is not started yet, as Hikari only starts it on the first
49   * connection, is left out until then.
50   *
51   * <p>The health is enabled on its own, as it lets systemd restart the application: while disabled, the application is always healthy. When enabled, it is
52   * unhealthy while any pool is starved: all its connections are in use, and threads are waiting for one, naming the starved data sources in the health
53   * details, under {@code starvedDataSources}. A starved pool usually means connections leak, as code that takes a connection never gives it back, which
54   * only a restart recovers from. As a burst of requests can also starve a pool for a while, the unhealthy state should be delayed, with
55   * {@link #setUnhealthyDelay(java.time.Duration)}: the pool must then stay starved, every time the health is evaluated, for that long.
56   *
57   * <p>A database that cannot be reached does not starve the pools: their threads wait for a connection, but the connections are not in use, as the pools
58   * cannot open them. That is left for the health indicators of Spring Boot Actuator.
59   *
60   * @author Joao Silva
61   */
62  public class HikariStatusProvider extends DelayableHealthProvider implements StatusProvider, SmartInitializingSingleton {
63  
64    private final ConfigurableListableBeanFactory beanFactory;
65  
66    private final boolean health;
67  
68    // set once all singleton beans are created, read by the status updates and the health evaluations
69    private volatile Map<String, HikariDataSource> pools = Map.of();
70  
71    /**
72     * Creates a new instance that finds the data sources in the given bean factory.
73     *
74     * @param beanFactory the bean factory to find the data sources in
75     * @param health <code>true</code> to report the application as unhealthy while any pool is starved
76     */
77    public HikariStatusProvider(ConfigurableListableBeanFactory beanFactory, boolean health) {
78      this.beanFactory = requireNonNull(beanFactory, "Bean factory must not be null");
79      this.health = health;
80    }
81  
82    // for tests: whether a starved pool reports the application as unhealthy
83    boolean healthEnabled() {
84      return health;
85    }
86  
87    /**
88     * Finds the HikariCP connection pools among the data source beans that are already created.
89     */
90    @Override
91    public void afterSingletonsInstantiated() {
92      Map<String, HikariDataSource> found = new LinkedHashMap<>();
93      // a pool wrapped by several data source beans, such as a proxy that is also a bean, is only reported once, under the first name
94      Set<HikariDataSource> seen = Collections.newSetFromMap(new IdentityHashMap<>());
95      createdBeans(beanFactory, DataSource.class).forEach((name, dataSource) -> {
96        HikariDataSource pool = DataSourceUnwrapper.unwrap(dataSource, HikariConfigMXBean.class, HikariDataSource.class);
97        if (pool != null && seen.add(pool)) {
98          found.put(name, pool);
99        }
100     });
101     pools = Collections.unmodifiableMap(found);
102   }
103 
104   @Override
105   public String status() {
106     List<String> statuses = new ArrayList<>();
107     pools.forEach((name, pool) -> {
108       // null until the pool is started
109       HikariPoolMXBean mxBean = pool.getHikariPoolMXBean();
110       if (mxBean != null) {
111         statuses.add(String.format(Locale.ROOT, "%s: %d/%d connections, %d waiting", name, mxBean.getActiveConnections(), pool.getMaximumPoolSize(),
112             mxBean.getThreadsAwaitingConnection()));
113       }
114     });
115     return String.join(", ", statuses);
116   }
117 
118   /**
119    * Provides the health of the application, without the delay: unhealthy while any pool is starved, if enabled.
120    *
121    * @return the health of the application
122    */
123   @Override
124   protected Health evaluateHealth() {
125     if (!health) {
126       return Health.HEALTHY;
127     }
128     List<String> starved = new ArrayList<>();
129     pools.forEach((name, pool) -> {
130       HikariPoolMXBean mxBean = pool.getHikariPoolMXBean();
131       if (mxBean != null && mxBean.getThreadsAwaitingConnection() > 0 && mxBean.getActiveConnections() >= pool.getMaximumPoolSize()) {
132         starved.add(name);
133       }
134     });
135     return starved.isEmpty() ? Health.HEALTHY : new Health(false, Map.of("starvedDataSources", starved));
136   }
137 }
