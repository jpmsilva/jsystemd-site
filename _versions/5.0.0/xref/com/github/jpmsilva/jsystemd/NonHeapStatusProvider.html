17  package com.github.jpmsilva.jsystemd;
18  
19  import static com.github.jpmsilva.jsystemd.SystemdUtilities.formatByteCount;
20  
21  import java.lang.management.ManagementFactory;
22  import java.lang.management.MemoryMXBean;
23  import java.util.Locale;
24  import java.util.Optional;
25  
26  /**
27   * Implementation of {@link StatusProvider} that provides the memory used outside of the heap by the JVM, such as by the metaspace and the code cache, out
28   * of its committed size, such as {@code Non-heap: 62.7 MiB/64.1 MiB}.
29   *
30   * <p>The committed size is the memory the JVM currently holds for it, which it grows as needed: it is not a limit, as the non-heap memory usually has none.
31   * The direct and mapped buffers are not part of it, and are reported by the {@link BufferPoolStatusProvider}.
32   *
33   * @author Joao Silva
34   */
35  public class NonHeapStatusProvider implements StatusProvider {
36  
37    /**
38     * Create a new NonHeapStatusProvider.
39     */
40    public NonHeapStatusProvider() {
41    }
42  
43    @Override
44    public String status() {
45      return Optional.ofNullable(ManagementFactory.getMemoryMXBean())
46          .map(MemoryMXBean::getNonHeapMemoryUsage)
47          .map(t -> String.format(Locale.ROOT, "Non-heap: %s/%s", formatByteCount(t.getUsed()), formatByteCount(t.getCommitted())))
48          .orElse("");
49    }
50  }
