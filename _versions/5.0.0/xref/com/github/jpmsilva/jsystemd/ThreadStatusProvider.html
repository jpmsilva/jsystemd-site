17  package com.github.jpmsilva.jsystemd;
18  
19  import java.lang.management.ManagementFactory;
20  import java.lang.management.ThreadInfo;
21  import java.lang.management.ThreadMXBean;
22  import java.util.ArrayList;
23  import java.util.Arrays;
24  import java.util.EnumMap;
25  import java.util.List;
26  import java.util.Locale;
27  import java.util.Map;
28  import java.util.Objects;
29  import org.jspecify.annotations.Nullable;
30  
31  /**
32   * Provides information regarding the live threads: as a {@link StatusProvider}, the number of threads by state, and the number of deadlocked
33   * threads, if any, for the systemd status, and as a {@link HealthProvider}, for the systemd watchdog, reporting the application as unhealthy under the
34   * health conditions it is configured with.
35   *
36   * <p>Only platform threads are reported: virtual threads are not visible through {@link ThreadMXBean}.
37   *
38   * <p>Each health condition is enabled on its own, and the application is healthy while none of the enabled ones holds, which is always the case when
39   * none is enabled. The health conditions are:
40   * <ul>
41   *   <li>deadlock health: unhealthy while any thread is deadlocked, naming the deadlocked threads in the health details, under
42   *   {@code deadlockedThreads}. Requires deadlock detection. As a deadlock never resolves by itself, the unhealthy state is not delayed, unlike the
43   *   {@link DelayableHealthProvider} ones: delaying the restart would only delay the recovery.</li>
44   * </ul>
45   *
46   * <p>Both reading the thread states and detecting deadlocks pause the JVM briefly at a safepoint, for a time that grows with the number of threads.
47   * This is negligible at the usual status update and watchdog periods.
48   *
49   * @author Joao Silva
50   */
51  public class ThreadStatusProvider implements StatusProvider, HealthProvider {
52  
53    @Nullable
54    private final ThreadMXBean threads;
55  
56    private final boolean detectDeadlocks;
57  
58    private final boolean deadlockHealth;
59  
60    /**
61     * Create a new ThreadStatusProvider.
62     *
63     * @param detectDeadlocks <code>true</code> to also detect deadlocked threads, reporting them in the status
64     * @param deadlockHealth <code>true</code> to report the application as unhealthy while any thread is deadlocked, which requires detecting deadlocks
65     * @throws IllegalArgumentException if deadlock health is enabled without detecting deadlocks
66     */
67    public ThreadStatusProvider(boolean detectDeadlocks, boolean deadlockHealth) {
68      this(ManagementFactory.getThreadMXBean(), detectDeadlocks, deadlockHealth);
69    }
70  
71    ThreadStatusProvider(@Nullable ThreadMXBean threads, boolean detectDeadlocks, boolean deadlockHealth) {
72      if (deadlockHealth && !detectDeadlocks) {
73        throw new IllegalArgumentException("Deadlock health requires detecting deadlocks");
74      }
75      this.threads = threads;
76      this.detectDeadlocks = detectDeadlocks;
77      this.deadlockHealth = deadlockHealth;
78    }
79  
80    // for tests: whether deadlocked threads are detected
81    boolean detectDeadlocks() {
82      return detectDeadlocks;
83    }
84  
85    // for tests: whether deadlocked threads report the application as unhealthy
86    boolean deadlockHealth() {
87      return deadlockHealth;
88    }
89  
90    @Override
91    public String status() {
92      ThreadMXBean threads = this.threads;
93      if (threads == null) {
94        return "";
95      }
96      // threads may terminate between listing them and reading their state, in which case their information is null
97      Map<Thread.State, Integer> counts = new EnumMap<>(Thread.State.class);
98      int total = 0;
99      for (ThreadInfo info : threads.getThreadInfo(threads.getAllThreadIds())) {
100       if (info != null) {
101         counts.merge(info.getThreadState(), 1, Integer::sum);
102         total++;
103       }
104     }
105 
106     // the statuses of all providers are joined with commas, so the details are separated otherwise
107     List<String> details = new ArrayList<>();
108     counts.forEach((state, count) -> details.add(String.format(Locale.ROOT, "%d %s", count, describe(state))));
109     int deadlocked = detectDeadlocks ? deadlockedThreads(threads).length : 0;
110     if (deadlocked > 0) {
111       details.add(String.format(Locale.ROOT, "%d deadlocked", deadlocked));
112     }
113     return String.format(Locale.ROOT, "Threads: %d (%s)", total, String.join("; ", details));
114   }
115 
116   private static String describe(Thread.State state) {
117     return state.name().toLowerCase(Locale.ROOT).replace('_', ' ');
118   }
119 
120   /**
121    * Reports the application as unhealthy while any of the enabled health conditions holds. Always healthy when none is enabled.
122    *
123    * @return the health of the application
124    */
125   @Override
126   public Health health() {
127     ThreadMXBean threads = this.threads;
128     if (threads == null || !deadlockHealth) {
129       return Health.HEALTHY;
130     }
131     long[] ids = deadlockedThreads(threads);
132     if (ids.length == 0) {
133       return Health.HEALTHY;
134     }
135     // threads may terminate between finding them and reading their names, in which case their information is null
136     List<String> names = Arrays.stream(threads.getThreadInfo(ids)).filter(Objects::nonNull).map(ThreadInfo::getThreadName).toList();
137     return new Health(false, Map.of("deadlockedThreads", names));
138   }
139 
140   private static long [] deadlockedThreads(ThreadMXBean threads) {
141     // deadlocks involving ownable synchronizers, such as ReentrantLock, can only be found when the JVM supports monitoring their usage
142     long[] ids = threads.isSynchronizerUsageSupported() ? threads.findDeadlockedThreads() : threads.findMonitorDeadlockedThreads();
143     // null when no thread is deadlocked
144     return ids == null ? new long[0] : ids;
145   }
146 }
