55  /**
56   * Integrates Keycloak with systemd, following the life cycle of Keycloak:
57   * <ol>
58   *   <li>when Keycloak initializes its providers, at the start of its bootstrap, the integration starts, and the status shows
59   *   {@code Startup: bootstrapping};</li>
60   *   <li>once Keycloak has completed its bootstrap, which includes migrating the database and importing realms, and its HTTP server listens, systemd is
61   *   told that Keycloak is ready. Either may happen last: Keycloak bootstraps once its HTTP server listens when the health endpoints are enabled, answering
62   *   503 until then, and before it starts its HTTP server otherwise. When the health endpoints are enabled, Keycloak is also only ready once its readiness
63   *   health checks, those of its {@code /health/ready} endpoint, are all up;</li>
64   *   <li>as soon as Keycloak is told to stop, systemd is told that it is stopping, and the status shows {@code Shutdown: stopping};</li>
65   *   <li>once Keycloak closes its providers, the integration stops.</li>
66   * </ol>
67   *
68   * <p>The watchdog heartbeats are sent when the service unit sets <code>WatchdogSec</code>, and the status shows the status of the JVM.
69   *
70   * @author Joao Silva
71   */
72  public class SystemdProviderFactory implements ProviderFactory<SystemdProvider> {

85    private static final String DEFAULT_STATUS_PROVIDERS = "heap,old-generation,cgroup-memory,threads,file-descriptors,non-heap,buffer-pools,class-loader";

87    private static final String STATUS_UPDATE_PERIOD_SECONDS = "status-update-period-seconds";
88    private static final int DEFAULT_STATUS_UPDATE_PERIOD_SECONDS = 5;
89    private static final String WATCHDOG_ENABLED = "watchdog-enabled";
90    private static final String STATUS_PROVIDERS_OPTION = "status-providers";

92    // the first and the longest wait between checks for readiness, once Keycloak has completed its bootstrap
93    private static final long FIRST_READY_CHECK_DELAY_MILLIS = 50;
94    private static final long MAX_READY_CHECK_DELAY_MILLIS = 500;

97    private final BooleanSupplier listening;

109   // held by ready() and stopping(), which run on different threads, so that one runs entirely before the other: ready() would otherwise clear the stage

111   // the startup is set, for the same reason
112   private final Lock lifecycleLock = new ReentrantLock();

138   private static Map<String, Supplier<StatusProvider>> knownStatusProviders() {
139     Map<String, Supplier<StatusProvider>> providers = new LinkedHashMap<>();
140     providers.put("heap", HeapStatusProvider::new);

142     providers.put("cgroup-memory", CgroupMemoryStatusProvider::new);

147     providers.put("non-heap", NonHeapStatusProvider::new);
148     providers.put("buffer-pools", BufferPoolStatusProvider::new);
149     providers.put("class-loader", ClassLoaderStatusProvider::new);
150     return providers;
151   }

164   /**
165    * Starts the integration with systemd, when Keycloak initializes its providers, at the start of its bootstrap.
166    *
167    * @param config the options of the factory
168    */
169   @Override
170   public void init(Config.Scope config) {
171     Systemd systemd = configure(config);
172     systemd.logStatus();
173     PhaseStatusProvider phase = new PhaseStatusProvider(true);
174     // the stage of the life cycle comes first; the other status providers are only asked when running under systemd
175     systemd.addStatusProviders(phase);
176     if (Systemd.isUnderSystemd()) {
177       systemd.addStatusProviders(statusProviders(config));
178     }
179     this.phase = phase;
180     phase.setPhase("Startup: bootstrapping");
181     // a shutdown hook runs as soon as the JVM is told to stop, before Keycloak starts shutting down, which only closes the providers later
182     Thread stoppingHook = new Thread(this::stopping, "jsystemd-stopping");
183     Runtime.getRuntime().addShutdownHook(stoppingHook);
184     this.stoppingHook = stoppingHook;
185   }
186 
187   private static Systemd configure(Config.Scope config) {
188     Systemd systemd = Systemd.instance();
189     // nothing is scheduled when not running under systemd, as nothing would be sent
190     if (!Systemd.isUnderSystemd()) {
191       return systemd;
192     }
193     systemd.enableStatusUpdate(statusUpdatePeriod(config));
194     if (config.getBoolean(WATCHDOG_ENABLED, true)) {
195       systemd.enableWatchdog();
196     }
197     return systemd;
198   }
199 
200   /**
201    * Returns the period of the status updates, in the {@code status-update-period-seconds} option: one that is not positive is ignored, with a warning, as
202    * are unknown status providers, rather than failing the startup of Keycloak.
203    *
204    * @param config the configuration of the provider
205    * @return the period of the status updates
206    */
207   static Duration statusUpdatePeriod(Config.Scope config) {
208     int seconds = config.getInt(STATUS_UPDATE_PERIOD_SECONDS, DEFAULT_STATUS_UPDATE_PERIOD_SECONDS);
209     if (seconds < 1) {
210       logger.warn("Ignoring the status update period of {} seconds, which must be of at least a second, using {} seconds instead", seconds,
211           DEFAULT_STATUS_UPDATE_PERIOD_SECONDS);
212       seconds = DEFAULT_STATUS_UPDATE_PERIOD_SECONDS;
213     }
214     return Duration.ofSeconds(seconds);
215   }
216 
217   /**
218    * Returns the status providers that are enabled, in the order they are shown.
219    *
220    * @param config the options of the factory
221    * @return the status providers that are enabled
222    */
223   static List<StatusProvider> statusProviders(Config.Scope config) {
224     String[] configured = config.getArray(STATUS_PROVIDERS_OPTION);
225     List<String> enabled = Arrays.stream(configured == null ? DEFAULT_STATUS_PROVIDERS.split(",") : configured).map(String::trim).toList();
226     enabled.stream().filter(key -> !STATUS_PROVIDERS.containsKey(key))
227         .forEach(key -> logger.warn("Ignoring unknown status provider \"{}\", known ones are {}", key, STATUS_PROVIDERS.keySet()));
228     List<StatusProvider> providers = new ArrayList<>();
229     STATUS_PROVIDERS.forEach((key, provider) -> {
230       if (enabled.contains(key)) {
231         providers.add(provider.get());
232       }
233     });
234     return providers;
235   }

237   /**
238    * Tells systemd that Keycloak is ready once it has completed its bootstrap, and its HTTP server listens.
239    *
240    * @param factory the session factory of Keycloak
241    */
242   @Override
243   public void postInit(KeycloakSessionFactory factory) {
244     // published at the end of the bootstrap, right before Keycloak considers its bootstrap completed
245     factory.register(event -> {
246       if (event instanceof PostMigrationEvent) {
247         bootstrapCompleted();
248       }
249     });
250   }
251 
252   /**
253    * Tells systemd that Keycloak is ready, once its readiness health checks are all up when it has them, and once its HTTP server listens.
254    */
255   void bootstrapCompleted() {
256     Checks checks = healthReadiness.get();
257     if (checks == null) {
258       logger.info("Telling systemd that Keycloak is ready once it has completed its bootstrap and its HTTP server listens");
259       if (listening.getAsBoolean()) {
260         ready();
261       } else {
262         // the bootstrap runs while Quarkus starts, which only starts the HTTP server once it completes
263         readyWhen(this::waitingForTheHttpServer);
264       }
265     } else {
266       // the checks are not up yet, as Keycloak only considers its bootstrap completed once the event has been published
267       logger.info("Telling systemd that Keycloak is ready once its readiness health checks are all up");
268       readyWhen(() -> {
269         String waiting = waitingForTheHttpServer();
270         return waiting != null ? waiting : waitingFor(checks);
271       });
272     }
273   }
274 
275   private @Nullable String waitingForTheHttpServer() {
276     return listening.getAsBoolean() ? null : "the HTTP server";
277   }
278 
279   private static @Nullable String waitingFor(Checks checks) {
280     List<String> down = checks.down();
281     if (down == null) {
282       return null;
283     }
284     return "the readiness health checks" + (down.isEmpty() ? "" : " (down: " + String.join(", ", down) + ")");
285   }
286 
287   // the waiting is done on a thread of its own, so that neither the bootstrap nor the startup of Quarkus wait for it
288   private void readyWhen(Supplier<@Nullable String> waitingFor) {
289     Thread waiting = new Thread(() -> waitUntilReady(waitingFor), "jsystemd-ready");
290     waiting.setDaemon(true);
291     waiting.start();
292   }
293 
294   // waits until there is nothing left to wait for, showing what Keycloak waits for in the stage of the startup, so that a startup that does not
295   // complete, which systemd times out, tells why
296   private void waitUntilReady(Supplier<@Nullable String> waitingFor) {
297     Systemd systemd = Systemd.instance();
298     long delay = FIRST_READY_CHECK_DELAY_MILLIS;
299     String shown = null;
300     try {
301       // neither Quarkus nor the health checks have events that a provider can observe; the wait ends if Keycloak stops instead, which ends the
302       // startup, and so before the checks can no longer run
303       String waiting;
304       do {
305         Thread.sleep(delay);
306         delay = Math.min(delay * 2, MAX_READY_CHECK_DELAY_MILLIS);
307         if (systemd.isReady()) {
308           return;
309         }
310         waiting = waitingFor.get();
311         // only when it changes, as each change of the stage sends the status
312         if (waiting != null && !waiting.equals(shown)) {
313           shown = waiting;
314           startupStage("Startup: waiting for " + waiting);
315         }
316       } while (waiting != null);
317       ready();
318     } catch (InterruptedException e) {
319       Thread.currentThread().interrupt();
320     }
321   }
322 
323   // sets the stage of the startup, unless Keycloak is ready or stopping, holding the lock of ready() and stopping(), so as not to replace their stages
324   private void startupStage(String stage) {
325     PhaseStatusProvider phase = this.phase;
326     lifecycleLock.lock();
327     try {
328       if (phase != null && !Systemd.instance().isReady()) {
329         phase.setPhase(stage);
330       }
331     } finally {
332       lifecycleLock.unlock();
333     }
334   }
335 
336   /**
337    * Tells systemd that Keycloak is ready. Does nothing once Keycloak is stopping, nor if already ready.
338    */
339   void ready() {
340     Systemd systemd = Systemd.instance();
341     PhaseStatusProvider phase = this.phase;
342     lifecycleLock.lock();
343     try {
344       // the stage is only cleared by the call that tells systemd, so that a stopping stage is never cleared
345       if (phase != null && !systemd.isReady()) {
346         phase.clearPhase();
347         systemd.ready();
348       }
349     } finally {
350       lifecycleLock.unlock();
351     }
352   }
353 
354   /**
355    * Tells systemd that Keycloak is stopping. From then on, no more watchdog heartbeats are sent.
356    */
357   void stopping() {
358     PhaseStatusProvider phase = this.phase;
359     lifecycleLock.lock();
360     try {
361       if (phase != null) {
362         // heartbeats are stopped before anything else, see Systemd.stopping()
363         Systemd.instance().stopping();
364         phase.setPhase("Shutdown: stopping");
365       }
366     } finally {
367       lifecycleLock.unlock();
368     }
369   }

371   /**
372    * Stops the integration with systemd, once Keycloak closes its providers, telling systemd that Keycloak is stopping if not done before.
373    */
374   @Override
375   public void close() {
376     Thread stoppingHook = this.stoppingHook;
377     if (stoppingHook != null) {
378       try {
379         Runtime.getRuntime().removeShutdownHook(stoppingHook);
380       } catch (IllegalStateException e) {
381         // the JVM is shutting down, and has already run the hook, or is running it
382         logger.trace("Not removing the shutdown hook, as the JVM is shutting down", e);
383       }
384     }
385     if (phase != null) {
386       try {
387         Systemd.instance().close();
388       } catch (InterruptedException e) {
389         Thread.currentThread().interrupt();
390       }
391     }
392   }
