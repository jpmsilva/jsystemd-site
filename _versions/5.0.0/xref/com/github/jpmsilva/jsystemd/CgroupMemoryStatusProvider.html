17  package com.github.jpmsilva.jsystemd;
18  
19  import static com.github.jpmsilva.jsystemd.SystemdUtilities.formatByteCount;
20  
21  import java.io.IOException;
22  import java.nio.charset.StandardCharsets;
23  import java.nio.file.Files;
24  import java.nio.file.Path;
25  import java.util.List;
26  import java.util.Locale;
27  import org.jspecify.annotations.Nullable;
28  
29  /**
30   * Implementation of {@link StatusProvider} that provides the memory used by the control group of the process, out of its limit, such as
31   * {@code Cgroup memory: 700.0 MiB/1.0 GiB}, or without the limit when there is none, such as {@code Cgroup memory: 700.0 MiB}.
32   *
33   * <p>Unlike the memory reported by the other status providers, this is the memory the kernel accounts to the service, and holds it to: all the memory of
34   * the process, including the native memory of the JVM, the thread stacks and the page cache. Once it reaches the limit, the kernel kills the process, even
35   * though the heap may still have room. The limit is the one systemd sets with {@code MemoryMax} in the service unit, or in any of the slices it belongs
36   * to, whichever is lowest.
37   *
38   * <p>Only the unified hierarchy of control groups (cgroup v2) is supported, the one systemd uses on current Linux distributions, as the legacy one is
39   * deprecated. Nothing is reported otherwise, or when not running on Linux.
40   *
41   * @author Joao Silva
42   */
43  public class CgroupMemoryStatusProvider implements StatusProvider {
44  
45    private static final Path PROC_SELF_CGROUP = Path.of("/proc/self/cgroup");
46  
47    private static final Path CGROUP_ROOT = Path.of("/sys/fs/cgroup");
48  
49    private final Path root;
50  
51    // the control group of the process, or null if it is not in the unified hierarchy
52    @Nullable
53    private final Path cgroup;
54  
55    /**
56     * Create a new CgroupMemoryStatusProvider, for the control group the process is in when created.
57     */
58    public CgroupMemoryStatusProvider() {
59      this(PROC_SELF_CGROUP, CGROUP_ROOT);
60    }
61  
62    CgroupMemoryStatusProvider(Path procSelfCgroup, Path root) {
63      this.root = root.normalize();
64      this.cgroup = cgroup(procSelfCgroup, this.root);
65    }
66  
67    // for tests: the directory of the control group of the process, or null if it is not in the unified hierarchy
68    @Nullable
69    Path cgroup() {
70      return cgroup;
71    }
72  
73    private static @Nullable Path cgroup(Path procSelfCgroup, Path root) {
74      List<String> lines;
75      try {
76        lines = Files.readAllLines(procSelfCgroup, StandardCharsets.UTF_8);
77      } catch (IOException | SecurityException e) {
78        return null;
79      }
80      // the unified hierarchy is the one with id 0 and no controllers, such as "0::/system.slice/myservice.service"
81      for (String line : lines) {
82        if (line.startsWith("0::/")) {
83          Path cgroup = root.resolve(line.substring("0::/".length())).normalize();
84          // the memory controller may not be enabled for the group, or the unified hierarchy may only be mounted elsewhere, as in the hybrid layout
85          return cgroup.startsWith(root) && Files.isReadable(cgroup.resolve("memory.current")) ? cgroup : null;
86        }
87      }
88      return null;
89    }
90  
91    @Override
92    public String status() {
93      Path cgroup = this.cgroup;
94      if (cgroup == null) {
95        return "";
96      }
97      try {
98        long current = Long.parseLong(read(cgroup.resolve("memory.current")));
99        long limit = limit(cgroup);
100       return limit < 0 ? String.format(Locale.ROOT, "Cgroup memory: %s", formatByteCount(current))
101           : String.format(Locale.ROOT, "Cgroup memory: %s/%s", formatByteCount(current), formatByteCount(limit));
102     } catch (IOException | RuntimeException e) {
103       // the control group may be gone, such as when the process was moved to another one
104       return "";
105     }
106   }
107 
108   /**
109    * Returns the lowest limit of the control group and of its parents, as each of them limits the memory of all its children, or -1 if there is none.
110    *
111    * <p>The root of the hierarchy is included: the actual root group has no limit, and no {@code memory.max} file either, but in a container with a control
112    * group namespace of its own, as by default with Docker and Podman, the root seen from the container is the control group of the container, which has the
113    * limit of the container. The process is then in the root, as {@code 0::/}, or in one of its children when systemd runs in the container.
114    */
115   private long limit(Path cgroup) throws IOException {
116     long limit = -1;
117     for (Path group = cgroup; group != null && group.startsWith(root); group = group.getParent()) {
118       Path file = group.resolve("memory.max");
119       if (Files.isReadable(file)) {
120         String value = read(file);
121         if (!"max".equals(value)) {
122           long groupLimit = Long.parseLong(value);
123           limit = limit < 0 ? groupLimit : Math.min(limit, groupLimit);
124         }
125       }
126     }
127     return limit;
128   }
129 
130   private static String read(Path file) throws IOException {
131     return Files.readString(file, StandardCharsets.UTF_8).trim();
132   }
133 }
